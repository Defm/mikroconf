:local sysname [/system identity get name]
:local scriptname "doPushStatsToInfluxDB"
:global globalScriptBeforeRun
$globalScriptBeforeRun $scriptname

:global globalNoteMe
:global globalTgMessage
:global globalSecret
:global globalCallFetch
:local itsOk true
:local state ""

# a part of queue comment to locate queues to be processed
:local queueCommentMark "dtq"

# influxDB service URL (beware about port when /fetch)
:local tUrl "http://nas.home/api/v2/write\?bucket=httpapi&org=home"
:local tPingUrl "http://nas.home/ping"

# Configure at runtime: INFLUXDB_USER, INFLUXDB_PASSWORD and INFLUXDB_TOKEN in $globalSecret.
:local influxUser "$[$globalSecret get INFLUXDB_USER]"
:local influxPassword "$[$globalSecret get INFLUXDB_PASSWORD]"
:local influxToken "$[$globalSecret get INFLUXDB_TOKEN]"
:local authHeader "Authorization: Token $influxToken"

:local useBandwidthTest false

# preflight: service must be online, otherwise quiet exit (no Telegram spam)
:onerror errorDescr in={
    :set state "Checking if INFLUXDB Service online"
    $globalNoteMe value=$state
    :local result [/tool fetch http-method=get port=8086 user=$influxUser password=$influxPassword mode=http url="$tPingUrl" as-value output=user]
    $globalNoteMe value="Done"
} do={
    :set state "WARNING: INFLUXDB service unreachable ($errorDescr) - skipping stats push!"
    :log error $state
    $globalNoteMe value=$state
    :error $state
}

:if ($useBandwidthTest) do={

    :local txAvg 0
    :local rxAvg 0

    :local btServer 192.168.97.1
    # Configure at runtime: BANDWIDTH_TEST_USER and BANDWIDTH_TEST_PASSWORD in $globalSecret.
    :local btUser "$[$globalSecret get BANDWIDTH_TEST_USER]"
    :local btPassword "$[$globalSecret get BANDWIDTH_TEST_PASSWORD]"

    :set state "Starting VPN bandwidth test"
    $globalNoteMe value=$state

    /tool bandwidth-test protocol=tcp direction=transmit user=$btUser password=$btPassword address=$btServer duration=15s do={
        :set txAvg ($"tx-total-average" / 1048576)
    }

    /tool bandwidth-test protocol=tcp direction=receive user=$btUser password=$btPassword address=$btServer duration=15s do={
        :set rxAvg ($"rx-total-average" / 1048576)
    }

    :local fetchCmd "/tool fetch http-method=post port=8086 mode=http url=\"$tUrl\" http-header-field=\"$authHeader, content-type: text/plain\" http-data=\"bandwidth,host=$sysname,target=CHR transmit=$txAvg,receive=$rxAvg\" keep-result=no"
    $globalCallFetch $fetchCmd
}

/queue simple
:foreach z in=[find where comment~"$queueCommentMark"] do={

    :onerror errorDescr in={

        :local queueComment [get $z comment]
        :local ip [get $z target]

        :set state "Locating queue target IP for queue $queueComment"
        $globalNoteMe value=$state
        :if ([:typeof ($ip->0)] = "nil") do={
            :set state "Cant locate queue target IP for queue $queueComment. Skip it."
            $globalNoteMe value=$state
            :error "skip-continue"
        }
        :set ip ($ip->0)
        :set ip [:pick $ip 0 [:find $ip "/" -1]]
        $globalNoteMe value="Done"

        :local bytes [get $z bytes]
        :local upload [:pick $bytes 0 [:find $bytes "/"]]
        :local download [:pick $bytes ([:find $bytes "/"] + 1) [:len $bytes]]

        # dhcp server for IP->name translation
        :set state "Picking host name for $ip via DHCP"
        $globalNoteMe value=$state
        :local hostName ""
        :onerror errorDescr in={
            :set hostName [/ip dhcp-server lease get [find (address=$ip)] host-name]
        } do={
            :set state "Cant pick host name for $ip ($errorDescr). Skip it."
            $globalNoteMe value=$state
            :error "skip-continue"
        }

        :local typeOfValue [:typeof $hostName]
        :if (($typeOfValue = "nothing") or ($typeOfValue = "nil") or ([:len $hostName] = 0)) do={
            :set state "Got empty host name. Skip it."
            $globalNoteMe value=$state
            :error "skip-continue"
        }
        :set state "Got it $hostName"
        $globalNoteMe value=$state

        :set state "Pushing stats to influxDB about $hostName: UP = $upload, DOWN=$download"
        $globalNoteMe value=$state
        /tool fetch http-method=post port=8086 mode=http url="$tUrl" http-header-field="$authHeader, content-type: text/plain" http-data="traffic,host=$sysname,target=$hostName upload=$upload,download=$download" keep-result=no
        $globalNoteMe value="Done"

        :set state "Flushing stats.."
        $globalNoteMe value=$state
        reset-counters $z

    } do={
        :if ($errorDescr != "skip-continue") do={
            :set state "Error When $state: $errorDescr"
            $globalNoteMe value=$state
            :set itsOk false
        }
    }
}

:local inf ""
:if ($itsOk) do={
    :set inf "✓ SUCCESS $scriptname on $sysname: stats pushed"
}

:if (!$itsOk) do={
    :set inf "✗ FAILED $scriptname on $sysname: $state"
}

$globalNoteMe value=$inf

:if (!$itsOk) do={
    $globalTgMessage value=$inf
    :error $inf
}
