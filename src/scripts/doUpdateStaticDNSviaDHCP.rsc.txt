:local sysname [/system identity get name]
:local scriptname "doUpdateStaticDNSviaDHCP"
:global globalScriptBeforeRun
$globalScriptBeforeRun $scriptname

:global globalNoteMe
:global globalTgMessage
:local itsOk true
:local state ""

# Creates static DNS entries for DHCP clients in the named DHCP server.
# Hostnames passed to DHCP are appended with the zone.

# Name of the DHCP server instance:
:local dhcpServer "main-dhcp-server"

# DNS zone suffix:
:local dnsSuffix ".home"

# DNS TTL:
:local ttl "00:05:00"

# Enable console debug:
#:local debugout do={ :put ("DEBUG: " . [:tostr $1]) }
# Disable console debug:
:local debugout do={ :log debug $1 }

:local cleanHostname do={
    :local max ([:len $1] - 1)
    :if ($1 ~ "^[a-zA-Z0-9]+[a-zA-Z0-9\\-]*[a-zA-Z0-9]+\$" && ([:pick $1 ($max)] != "\00")) do={
        :return ($1)
    } else={
        :local cleaned ""
        :for i from=0 to=$max do={
            :local c [:pick $1 $i]
            :if ($c ~ "^[a-zA-Z0-9]{1}\$") do={
                :set cleaned ($cleaned . $c)
            } else={
                :if ($c = "-" and $i > 0 and $i < $max) do={
                    :set cleaned ($cleaned . $c)
                }
            }
        }
        :return ($cleaned)
    }
}

:onerror errorDescr in={

    :set state "Caching DHCP leases for $dhcpServer"
    $globalNoteMe value=$state

    :local dhcpLeases
    :set dhcpLeases [:toarray ""]
    /ip dhcp-server lease
    :foreach lease in=[find where server=$dhcpServer] do={
        :local hostRaw [get $lease host-name]
        :if ([:len $hostRaw] > 0) do={
            :local hostCleaned
            :set hostCleaned [$cleanHostname $hostRaw]
            :set ($dhcpLeases->$hostCleaned) $lease
        }
    }

    :set state "Updating stale DNS entries"
    $globalNoteMe value=$state

    /ip dns static
    :foreach record in=[find where comment="<AUTO:DHCP:$dhcpServer>"] do={
        :local fqdn [get $record name]
        :local hostname [:pick $fqdn 0 ([:len $fqdn] - [:len $dnsSuffix])]
        :local leaseMatch ($dhcpLeases->$hostname)

        :if ([:len $leaseMatch] < 1) do={
            $debugout ("Removing stale DNS record '$fqdn'")
            remove $record
        } else={
            :local lease [/ip dhcp-server lease get $leaseMatch address]
            :if ($lease != [get $record address]) do={
                $debugout ("Updating stale DNS record '$fqdn' to $lease")
                :onerror errorDescr in={
                    set $record address=$lease
                } do={
                    :set state "Unable to update stale DNS record '$fqdn': $errorDescr"
                    $globalNoteMe value=$state
                }
            }
        }
    }

    :set state "Adding new DNS entries"
    $globalNoteMe value=$state

    /ip dns static
    :foreach k,v in=$dhcpLeases do={
        :local fqdn ($k . $dnsSuffix)
        :if ([:len [find where name=$fqdn]] < 1) do={
            :local lease [/ip dhcp-server lease get $v address]
            $debugout ("Creating DNS record '$fqdn': $lease")
            :onerror errorDescr in={
                add name=$fqdn address=$lease ttl=$ttl comment="<AUTO:DHCP:$dhcpServer>"
            } do={
                :set state "Unable to create DNS record '$fqdn': $errorDescr"
                $globalNoteMe value=$state
            }
        }
    }

    $globalNoteMe value="Done"

} do={
    :set state "Error When $state: $errorDescr"
    $globalNoteMe value=$state
    :set itsOk false
}

:local inf ""
:if ($itsOk) do={
    :set inf "✓ SUCCESS $scriptname on $sysname: static DNS synced via DHCP"
}

:if (!$itsOk) do={
    :set inf "✗ FAILED $scriptname on $sysname: $state"
}

$globalNoteMe value=$inf

:if (!$itsOk) do={
    $globalTgMessage value=$inf
    :error $inf
}
