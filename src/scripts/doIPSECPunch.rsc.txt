:local sysname [/system identity get name]
:local scriptname "doIPSECPunch"
:global globalScriptBeforeRun
$globalScriptBeforeRun $scriptname

:global globalNoteMe
:global globalTgMessage
:global globalPeerRetryCount
:local itsOk true
:local state ""

# IPSec/L2TP Connection Monitor & Auto-Recovery
# every IPSec policy is examined: L2TP host-tunnel, specific routes, ping; failed peers are recovered (kill L2TP session / IPSec peers)

:local config {
    "pingCount"=10;
    "pingSuccessRate"=80;
    "pingInterval"=1000ms;
    "pingRetryDelay"=20s;
    "maxRetryAttempts"=3;
    "flushHour"=23;
    "enableAutoRecovery"=true
}

:local pingTrigger (($config->"pingCount") * ($config->"pingSuccessRate") / 100)

# Example call
# :local calc [$ipCalc "192.168.1.0/24"]; :put ($calc->"hostmin")
# calculate IP details from CIDR
:local ipCalc do={
    :local cidr [:tostr $1]
    :local address [:toip [:pick $cidr 0 [:find $cidr "/"]]]
    :local bits [:tonum [:pick $cidr ([:find $cidr "/"] + 1) [:len $cidr]]]
    :local mask ((255.255.255.255 << (32 - $bits)) & 255.255.255.255)
    :return {
        "network"=($address & $mask);
        "hostmin"=(($address & $mask) | 0.0.0.1);
        "broadcast"=($address | ~$mask)
    }
}

# Example call
# :local ip [$extractIp "10.0.0.1/32"]
# extract first IP from CIDR or return as-is
:local extractIp do={
    :local input [:tostr $1]
    :if ([:find $input "/"] >= 0) do={
        :return [:pick $input 0 [:find $input "/"]]
    }
    :return $input
}

# Example call
# :local l2tpInfo [$findL2tpSession $dstAddr]
# find L2TP session info (incoming server session or outgoing client)
:local findL2tpSession do={
    :local dstIp [:tostr $1]
    :local result {"found"=false; "type"="none"}

    # Incoming L2TP (server)
    :local incoming [/ppp/active find caller-id=$dstIp]
    :if ([:len $incoming] != 0) do={
        :set ($result->"running") true
        :set ($result->"found") true
        :set ($result->"type") "incoming"
        :set ($result->"session") $incoming
        :set ($result->"peer") [/ppp/active get $incoming name]
        :set ($result->"remoteIP") [/ppp/active get $incoming address]
        :local network [/ip/address get [find network=($result->"remoteIP")] address]
        :set ($result->"localIP") [:pick $network 0 [:find $network "/"]]
        :return $result
    }

    # Outgoing L2TP (client)
    :local outgoing [/interface/l2tp-client find where !disabled connect-to=$dstIp]
    :if ([:len $outgoing] != 0) do={

        :local online [/interface/l2tp-client get $outgoing running]
        :set ($result->"running") $online
        :set ($result->"found") true
        :set ($result->"type") "outgoing"
        :set ($result->"session") $outgoing
        :set ($result->"peer") [/interface/l2tp-client get $outgoing name]

        :if ($online) do={
            :local monitor [/interface l2tp-client monitor $outgoing once as-value]
            :set ($result->"remoteIP") ($monitor->"remote-address")
            :local network [/ip/address get [find network=($result->"remoteIP")] address]
            :set ($result->"localIP") [:pick $network 0 [:find $network "/"]]
        } else={
            :set ($result->"remoteIP") 0.0.0.1
            :set ($result->"localIP") 0.0.0.1
        }

        :return $result
    }

    :return $result
}

# Example call
# :local pingResult [$testConnectivity $dstIp $srcIp 10 8 20s]
# test connectivity with one retry
:local testConnectivity do={
    :local targetIp [:toip $1]
    :local sourceIp [:toip $2]
    :local count [:tonum $3]
    :local threshold [:tonum $4]
    :local retryDelay $5

    :local result1 [/ping address=$targetIp src-address=$sourceIp count=$count interval=1000ms]
    :if ($result1 >= $threshold) do={
        :return {"success"=true; "attempts"=1; "received"=$result1; "required"=$threshold}
    }

    :delay $retryDelay
    :local result2 [/ping address=$targetIp src-address=$sourceIp count=$count interval=1000ms]
    :if ($result2 >= $threshold) do={
        :return {"success"=true; "attempts"=2; "received"=$result2; "required"=$threshold}
    }

    :return {"success"=false; "attempts"=2; "received"=$result2; "required"=$threshold}
}

# Example call
# :local routeCheck [$checkRoute $dstIp $srcIp]
# check route availability
:local checkRoute do={
    :local dstIp [:toip $1]
    :local srcIp [:toip $2]
    :local check [/ip/route/check dst-ip=$dstIp src-ip=$srcIp once as-value]
    :local status ($check->"status")
    :if ($status = "ok") do={
        :return {
            "reachable"=true;
            "interface"=($check->"interface");
            "gateway"=($check->"gateway");
            "nexthop"=($check->"nexthop")
        }
    }
    :return {"reachable"=false; "status"=$status}
}

# Example call
# :local killed [$killL2tpSession $l2tpInfo]
# predicate: kill L2TP session, returns true when something was killed/restarted
:local killL2tpSession do={
    :global globalNoteMe
    :local state ""
    :local sessionInfo $1
    :local killed false

    :if (!($sessionInfo->"found")) do={ :return $killed }

    :onerror errorDescr in={
        :if (($sessionInfo->"type") = "incoming") do={
            /ppp/active remove ($sessionInfo->"session")
            :set state ("Killed incoming L2TP session: " . ($sessionInfo->"peer"))
            $globalNoteMe value=$state
        } else={
            /interface/l2tp-client disable ($sessionInfo->"session")
            :delay 4s
            /interface/l2tp-client enable ($sessionInfo->"session")
            :set state ("Restarted outgoing L2TP client: " . ($sessionInfo->"peer"))
            $globalNoteMe value=$state
        }
        :set killed true
    } do={
        :set state ("FAILED to kill L2TP session for peer: " . ($sessionInfo->"peer") . ": $errorDescr")
        $globalNoteMe value=$state
        :log error $state
    }

    :return $killed
}

# Example call
# :local killed [$killIpsecPeers $peerIp]
# predicate: kill IPSec active peers, returns true when at least one peer was removed
:local killIpsecPeers do={
    :global globalNoteMe
    :local state ""
    :local peerIp [:tostr $1]
    :local activePeers [/ip/ipsec/active-peers find remote-address=$peerIp]

    :if ([:len $activePeers] = 0) do={
        :set state "No active IPSec peers found for $peerIp"
        $globalNoteMe value=$state
        :log warning $state
        :return false
    }

    :local killCount 0
    :foreach peer in=$activePeers do={
        :onerror errorDescr in={
            /ip/ipsec/active-peers remove $peer
            :set killCount ($killCount + 1)
        } do={
            :set state "FAILED to kill IPSec peer: $peerIp: $errorDescr"
            $globalNoteMe value=$state
            :log error $state
        }
    }

    :if ($killCount > 0) do={
        :set state "Killed $killCount IPSec active peer(s) for $peerIp"
        $globalNoteMe value=$state
        :return true
    }

    :return false
}

# Example call
# :local recovered [$recoverPeer $dstAddr $peerName $l2tpInfo $config $killL2tpSession $killIpsecPeers]
# predicate: perform recovery actions, returns true when recovery was attempted
:local recoverPeer do={
    :global globalNoteMe
    :global globalPeerRetryCount
    :local state ""

    :local peerIp [:tostr $1]
    :local peerName [:tostr $2]
    :local l2tpSession $3
    :local config $4
    :local killL2tp $5
    :local killIpsec $6

    :local retryCount ($globalPeerRetryCount->$peerIp)

    :if ([:typeof $retryCount] != "num") do={ :set retryCount 0 }

    :set retryCount ($retryCount + 1)
    :set ($globalPeerRetryCount->$peerIp) $retryCount

    :if ($retryCount > ($config->"maxRetryAttempts")) do={
        :set state "Peer $peerName ($peerIp) exceeded max retry attempts ($retryCount) - MANUAL INTERVENTION REQUIRED"
        $globalNoteMe value=$state
        :log error $state
        :return false
    }

    :set state "RECOVERY START: $peerName ($peerIp) - Attempt $retryCount"
    $globalNoteMe value=$state

    :if (($l2tpSession->"found")) do={
        :local killed [$killL2tp $l2tpSession]
        :if ($killed) do={ :delay 2s }
    }

    :local ipsecKilled [$killIpsec $peerIp]
    :if ($ipsecKilled) do={ :delay 2s }

    :set state "RECOVERY COMPLETE: $peerName ($peerIp)"
    $globalNoteMe value=$state
    :return true
}

:set state "IPSec/L2TP Connection Monitor Started"
$globalNoteMe value=$state

:local policiesOnline [/ip/ipsec/policy find (!template !disabled)]

:set state ("Found " . [:len $policiesOnline] . " IPSec policies to investigate")
$globalNoteMe value=$state

:local stats {
    "totalPolicies"=[:len $policiesOnline];
    "testedPeers"=0;
    "passedPeers"=0;
    "failedPeers"=0;
    "recoveredPeers"=0;
    "skippedPeers"=0
}

:foreach policy in=$policiesOnline do={

    :local peerName
    :local srcAddr
    :local dstAddr
    :local l2tpInfo

    # every failed check throws "skip-continue": checks for this policy stop and recovery runs in the handler
    :onerror errorDescr in={

        :set peerName [/ip/ipsec/policy get $policy peer]
        :local isTunnel [/ip/ipsec/policy get $policy tunnel]
        :local isActive [/ip/ipsec/policy get $policy active]
        :local polState [/ip/ipsec/policy get $policy ph2-state]

        :if ($isTunnel) do={
            :local dstSubnet [/ip/ipsec/policy get $policy dst-address]
            :local srcSubnet [/ip/ipsec/policy get $policy src-address]
            :local dstCalc [$ipCalc $dstSubnet]
            :local srcCalc [$ipCalc $srcSubnet]
            :set dstAddr ($dstCalc->"hostmin")
            :set srcAddr ($srcCalc->"hostmin")
        } else={
            :set dstAddr [$extractIp [/ip/ipsec/policy get $policy dst-address]]
            :set srcAddr [$extractIp [/ip/ipsec/policy get $policy src-address]]
        }

        :set ($stats->"testedPeers") (($stats->"testedPeers") + 1)

        :set state "Examine Peer $peerName ($dstAddr)"
        $globalNoteMe value=$state

        # 1) Looking for host L2TP session
        :set l2tpInfo [$findL2tpSession $dstAddr]

        :if (!$isActive) do={
            :set state "Non-active policy investigated for $peerName"
            $globalNoteMe value=$state
            :log error $state
            :set ($stats->"failedPeers") (($stats->"failedPeers") + 1)
            :error "skip-continue"
        }

        :if ($polState != "established") do={

            :set state "Non-established policy investigated for $peerName"
            $globalNoteMe value=$state
            :log error $state

            # some more diagnostic
            :if (($l2tpInfo->"found")) do={
                :if (!($l2tpInfo->"running")) do={
                    :set state ("Disconnected (Rx-Tx 0-0) " . ($l2tpInfo->"type") . " L2TP host-tunnel found for IPSEC: " . ($l2tpInfo->"peer"))
                    $globalNoteMe value=$state
                    :log error $state
                }
            }

            :set ($stats->"failedPeers") (($stats->"failedPeers") + 1)
            :error "skip-continue"
        }

        :local policyDstIp $dstAddr
        :local policySrcIp $srcAddr

        :local mustUseL2tp false

        :local l2tpIf ""
        :local l2tpRouteIf ""
        :local policyRouteIf ""

        :if (($l2tpInfo->"found")) do={

            :if (!($l2tpInfo->"running")) do={
                :set state ("Disconnected (Rx-Tx 0-0) " . ($l2tpInfo->"type") . " L2TP host-tunnel found for IPSEC: " . ($l2tpInfo->"peer"))
                $globalNoteMe value=$state
                :log error $state
                :set ($stats->"failedPeers") (($stats->"failedPeers") + 1)
                :error "skip-continue"
            }

            :set state ("Active " . ($l2tpInfo->"type") . " L2TP host-tunnel found for IPSEC: " . ($l2tpInfo->"peer"))
            $globalNoteMe value=$state

            :local l2tpDstAddr ($l2tpInfo->"remoteIP")
            :local l2tpSrcAddr ($l2tpInfo->"localIP")
            :set mustUseL2tp true
            :set l2tpIf ($l2tpInfo->"peer")

            :set state "Looking for specific (l2tp) routes to $l2tpDstAddr from $l2tpSrcAddr"
            $globalNoteMe value=$state

            # we need specific route in the main table to that subnet
            :local l2tpRouteCheck [$checkRoute $l2tpDstAddr $l2tpSrcAddr]
            :if (!($l2tpRouteCheck->"reachable")) do={
                :set state ("Cannot find specific route (l2tp) to $l2tpDstAddr from $l2tpSrcAddr - status: " . ($l2tpRouteCheck->"status"))
                $globalNoteMe value=$state
                :log error $state
                :set ($stats->"failedPeers") (($stats->"failedPeers") + 1)
                :error "skip-continue"
            }

            :set l2tpRouteIf ($l2tpRouteCheck->"interface")
            :set state ("Found specific route (l2tp) to $l2tpDstAddr from $l2tpSrcAddr via " . ($l2tpRouteCheck->"nexthop") . " ($l2tpRouteIf)")
            $globalNoteMe value=$state

            :set policyDstIp $l2tpDstAddr
            :set policySrcIp $l2tpSrcAddr

        } else={

            :if (!$isTunnel) do={
                # on transport mode l2tp session have to exist (just a local convenience)
                :set state "No active L2TP host-tunnel found for IPSEC peer - $peerName (on transport policy l2tp session have to exist)"
                $globalNoteMe value=$state
                :log error $state
                :set ($stats->"failedPeers") (($stats->"failedPeers") + 1)
                :error "skip-continue"
            }

            :set state "No active L2TP host-tunnel found for IPSEC peer - $peerName (pure IPSEC, tunneled=$isTunnel)"
            $globalNoteMe value=$state
        }

        # 2) Policy route check
        :set state "Looking for specific (policy) routes to $policyDstIp from $policySrcIp"
        $globalNoteMe value=$state

        # we need specific route in the main table to that subnet
        :local policyRouteCheck [$checkRoute $policyDstIp $policySrcIp]
        :if (!($policyRouteCheck->"reachable")) do={
            :set state ("Cannot find specific route (policy) to $policyDstIp from $policySrcIp - status: " . ($policyRouteCheck->"status"))
            $globalNoteMe value=$state
            :log error $state
            :set ($stats->"failedPeers") (($stats->"failedPeers") + 1)
            :error "skip-continue"
        }

        :set policyRouteIf ($policyRouteCheck->"interface")
        :set state ("Found specific route (policy) to $policyDstIp from $policySrcIp via " . ($policyRouteCheck->"nexthop") . " ($policyRouteIf)")
        $globalNoteMe value=$state

        # 3) both routes have to be via the same catched L2TP interface/gw
        :if ($mustUseL2tp) do={

            :set state "Checking policy-l2tp routes' interfaces match"
            $globalNoteMe value=$state

            :if (($policyRouteIf != $l2tpIf) or ($l2tpRouteIf != $l2tpIf)) do={
                :set state "Found routes mismatch (policy via $policyRouteIf, l2tp via $l2tpRouteIf, expected $l2tpIf)"
                $globalNoteMe value=$state
                :log error $state
                :set ($stats->"failedPeers") (($stats->"failedPeers") + 1)
                :error "skip-continue"
            }

            :set state "Found routes' interfaces are the same on L2TP $l2tpIf and Policy $peerName"
            $globalNoteMe value=$state

        } else={
            :set state "Skip policy-l2tp routes' interfaces match - no active L2TP host-tunnel available"
            $globalNoteMe value=$state
        }

        # 4) ping
        :set state "Pinging remote IP $policyDstIp"
        $globalNoteMe value=$state

        :local pingResult [$testConnectivity $policyDstIp $policySrcIp ($config->"pingCount") $pingTrigger ($config->"pingRetryDelay")]
        :if (($pingResult->"success")) do={

            :set state ("PASS: Ping test successful " . ($pingResult->"received") . "/" . ($pingResult->"required") . " (" . ($pingResult->"attempts") . " attempt(s))")
            $globalNoteMe value=$state
            :set ($stats->"passedPeers") (($stats->"passedPeers") + 1)

            # :set ($globalPeerRetryCount->$dstAddr) 0

        } else={

            :set state ("FAIL: Ping test failed " . ($pingResult->"received") . "/" . ($pingResult->"required") . " after " . ($pingResult->"attempts") . " attempts")
            $globalNoteMe value=$state
            :log error $state
            :set ($stats->"failedPeers") (($stats->"failedPeers") + 1)
            # going to recover
            :error "skip-continue"
        }

        :set state "Continue next"
        $globalNoteMe value=$state

    } do={

        :if (!($errorDescr ~ "^skip-continue")) do={
            :set state "Error When $state: $errorDescr"
            $globalNoteMe value=$state
            :log error $state
            :set itsOk false
        }

        # peer failed its checks (or crashed) - try to recover, then go next policy
        :if (($config->"enableAutoRecovery")) do={
            :local recovered [$recoverPeer $dstAddr $peerName $l2tpInfo $config $killL2tpSession $killIpsecPeers]
            :if ($recovered) do={
                :set ($stats->"recoveredPeers") (($stats->"recoveredPeers") + 1)
            }
        } else={
            :set state "Auto-recovery disabled - manual intervention required"
            $globalNoteMe value=$state
            :log warning $state
        }

        :set state "Continue next (in case of skip or errors)"
        $globalNoteMe value=$state
        :log warning $state
    }
}

# post-check validation
:set state "Waiting for system to come up after autorecovery"
$globalNoteMe value=$state

:delay 5s
:local policiesOnlineAfter [/ip/ipsec/policy find active ph2-state=established]
:local policiesOfflineAfter [/ip/ipsec/policy find (!template !disabled ph2-state!=established)]

# summary
:set state ("Total policies: " . ($stats->"totalPolicies"))
$globalNoteMe value=$state

:set state ("Tested: " . ($stats->"testedPeers") . " | Passed: " . ($stats->"passedPeers") . " | Failed: " . ($stats->"failedPeers"))
$globalNoteMe value=$state

:set state ("Recovered: " . ($stats->"recoveredPeers") . " | Skipped: " . ($stats->"skippedPeers"))
$globalNoteMe value=$state

:set state ("Policies after check: Online=" . [:len $policiesOnlineAfter] . " | Offline=" . [:len $policiesOfflineAfter])
$globalNoteMe value=$state

:local errorMessage ""

:if (($stats->"failedPeers") > 0) do={
    :set itsOk false
    :set errorMessage (($stats->"failedPeers") . " peer(s) failed connectivity test")
}

:if ([:len $policiesOfflineAfter] != 0) do={
    :set itsOk false
    :set errorMessage ($errorMessage . "; " . [:len $policiesOfflineAfter] . " policies offline after recovery")
}

:if (!$itsOk) do={
    :set state "IPSec/L2TP Monitor: FAILURES DETECTED - $errorMessage"
}

:local inf ""
:if ($itsOk) do={
    :set inf "✓ SUCCESS $scriptname on $sysname: IPSec/L2TP Monitor: all checks passed"
}

:if (!$itsOk) do={
    :set inf "✗ FAILED $scriptname on $sysname: $state"
}

$globalNoteMe value=$inf

:if (!$itsOk) do={
    $globalTgMessage value=$inf
    :error $inf
}
