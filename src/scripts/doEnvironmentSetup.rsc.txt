            :local context ""
            :if ([:len $2] > 0) do={ :set context " ($2 context)" }

            $globalNoteMe value=$state

        }
    }
}

# Example call
# $globalSafeScriptCall "doImperialMarch" "STARTUP"
# runs a script if it exists, never throws; $2 is an optional context postfix
# uses :log directly - it may be called before globalNoteMe exists
:global globalSafeScriptCall
:if (!any $globalSafeScriptCall) do={
    :global globalSafeScriptCall do={

        :local targetScript [:tostr $1]
        :local context ""
        :if ([:len $2] > 0) do={ :set context " ($2 context)" }
        :local itsOk false

        :if ([:len $targetScript] = 0) do={ :return $itsOk }

        :if ([:len [/system script find name=$targetScript]] = 0) do={
            :log warning "Script not found: $targetScript$context"
            :put "Script not found: $targetScript$context"
            :return $itsOk
        }

        :onerror errorDescr in={
            :log warning "Starting script: $targetScript$context"
            :put "Starting script: $targetScript$context"
            /system script run $targetScript
            :set itsOk true
        } do={
            :log error "FAIL Starting script: $targetScript$context: $errorDescr"
            :put "FAIL Starting script: $targetScript$context"
        }

        :return $itsOk
    }
}

# Example call
# $globalSecret
# $globalSecret get <name>
# $globalSecret set <name> password=<password>
# $globalSecret remove <name>
# $globalSecret print
# stored entries are marked in /ppp secret with comment "used by $SECRET" - this marker is data on devices, do not rename
:global globalSecret
:if (!any $globalSecret) do={
    :global globalSecret do={

        :global globalSecret

        # helpers
        :local fixprofile do={
            :if ([/ppp profile find name="null"]) do={ :put "nothing" } else={
                /ppp profile add bridge-learning=no change-tcp-mss=no local-address=0.0.0.0 name="null" only-one=yes remote-address=0.0.0.0 session-timeout=1s use-compression=no use-encryption=no use-mpls=no use-upnp=no
            }
        }

        :local lppp [:len [/ppp secret find where name=$2]]

        :local checkexist do={
            :if ($lppp = 0) do={
                :error "\$globalSecret: cannot find $2 in secret store"
            }
        }

        # $globalSecret
        :if ([:typeof $1] != "str") do={
            :put "\$globalSecret"
            :put "   uses /ppp/secrets to store stuff like REST apikeys, or other sensitive data"
            :put "\t\$globalSecret print - prints stored secret passwords"
            :put "\t\$globalSecret get <name> - gets a stored secret"
            :put "\t\$globalSecret set <name> password=\"YOUR_SECRET\" - sets a secret password"
            :put "\t\$globalSecret remove <name> - removes a secret"
            :return
        }

        # $globalSecret print
        :if ($1~"^pr") do={
            /ppp secret print where comment~"\\\$SECRET"
            :return
        }

        # $globalSecret get
        :if ($1~"get") do={
            $checkexist
            :local sid [/ppp secret find where name=$2]
            :if ([:len $sid] = 0) do={ :error "\$globalSecret: cannot find $2 in secret store" }
            :return [/ppp secret get $sid password]
        }

        # $globalSecret set
        :if ($1~"set|add") do={
            :if ([:typeof $password] = "str") do={} else={ :error "\$globalSecret: password= required" }
            :if ($lppp = 0) do={
                /ppp secret add name=$2 password=$password
            } else={
                /ppp secret set [/ppp secret find where name=$2] password=$password
            }
            $fixprofile
            /ppp secret set [/ppp secret find where name=$2] comment="used by \$SECRET"
            /ppp secret set [/ppp secret find where name=$2] profile="null"
            /ppp secret set [/ppp secret find where name=$2] service="async"
            :return [$globalSecret get $2]
        }

        # $globalSecret remove
        :if ($1~"rm|rem|del") do={
            $checkexist
            :return [/ppp secret remove [/ppp secret find where name=$2]]
        }

        :error "\$globalSecret: bad command"
    }
}

# Example call
# $globalNewNetworkMember ip=192.168.90.130 mac=50:DE:06:25:C2:FC gip=192.168.98.130 comm=iPadAlxPro ssid="WiFi 5"
:global globalNewNetworkMember
:if (!any $globalNewNetworkMember) do={
    :global globalNewNetworkMember do={

        :global globalNoteMe

        # to prevent connection
        :local guestDhcp "guest-dhcp-server"

        # to allow connection
        :local mainDhcp "main-dhcp-server"

        # when DHCP not using (add arp for leases)
        :local arpInterface "main-infrastructure-br"
        :local state "Adding new network member... "
        $globalNoteMe value=$state

        # incoming named params
        :local newIp [:tostr $ip]
        :local newBlockedIp [:tostr $gip]
        :local newMac [:tostr $mac]
        :local comment [:tostr $comm]
        :local newSsid [:tostr $ssid]
        :if ([:len $newIp] > 0) do={
            :if ([:typeof [:toip $newIp]] != "ip") do={
                :set state "Error: bad IP parameter passed - ($newIp)"
                $globalNoteMe value=$state
                :return false
            }
        } else={
            :set state "Error: bad IP parameter passed - ($newIp)"
            $globalNoteMe value=$state
            :return false
        }

        :onerror errorDescr in={

            :set state "Removing existing DHCP configuration for ($newIp/$newMac) on $mainDhcp"
            $globalNoteMe value=$state
            /ip dhcp-server lease remove [find address=$newIp]
            /ip dhcp-server lease remove [find mac-address=$newMac]

            :set state "Adding DHCP configuration for ($newIp/$newMac) on $mainDhcp"
            $globalNoteMe value=$state

            :if ([:len [/ip dhcp-server find where name="$mainDhcp"]] > 0) do={
                /ip dhcp-server lease add address=$newIp mac-address=$newMac server=$mainDhcp comment=$comment
                $globalNoteMe value="Done"
            } else={
                :set state "Cant find DHCP server $mainDhcp. SKIPPED."
                $globalNoteMe value=$state
            }

        } do={
            :set state "Error: something fail on DHCP configuration 'allow' step for ($newIp/$newMac) on $mainDhcp: $errorDescr"
            $globalNoteMe value=$state
            :return false
        }

        :onerror errorDescr in={

            /ip dhcp-server lease remove [find address=$newBlockedIp]
            :set state "Adding DHCP configuration for ($newBlockedIp/$newMac) on $guestDhcp (preventing connections to guest network)"
            $globalNoteMe value=$state

            :if ([:len [/ip dhcp-server find where name="$guestDhcp"]] > 0) do={
                /ip dhcp-server lease add address=$newBlockedIp block-access=yes mac-address=$newMac server=$guestDhcp comment=($comment . "(blocked)")
                $globalNoteMe value="Done"
            } else={
                :set state "Cant find DHCP server $guestDhcp. SKIPPED."
                $globalNoteMe value=$state
            }

        } do={
            :set state "Error: something fail on DHCP configuration 'block' step for ($newBlockedIp/$newMac) on $guestDhcp: $errorDescr"
            $globalNoteMe value=$state
            :return false
        }

        :onerror errorDescr in={

            :set state "Adding ARP static entries for ($newBlockedIp/$newMac) on $mainDhcp"
            $globalNoteMe value=$state
            /ip arp remove [find address=$newIp]
            /ip arp remove [find address=$newBlockedIp]
            /ip arp remove [find mac-address=$newMac]

            :if ([:len [/interface find where name="$arpInterface"]] > 0) do={
                /ip arp add address=$newIp interface=$arpInterface mac-address=$newMac comment=$comment
                $globalNoteMe value="Done"
            } else={
                :set state "Cant find interface $arpInterface. SKIPPED."
                $globalNoteMe value=$state
            }

        } do={
            :set state "Error: something fail on ARP configuration step: $errorDescr"
            $globalNoteMe value=$state
            :return false
        }

        :onerror errorDescr in={

            :set state "Adding CAPs ACL static entries for ($newBlockedIp/$newMac) on $newSsid"
            $globalNoteMe value=$state

            # avoid parse errors using Execute when no wireless package installed
            :if ([:len [/system package find where name="wireless" and disabled=no]] > 0) do={
                :local asyncCmd "/caps-man access-list remove [find mac-address=$newMac];"
                :local jobid [:execute script=$asyncCmd]

                :set asyncCmd "/caps-man access-list add action=accept allow-signal-out-of-range=10s client-to-client-forwarding=yes comment=\"$comment\" disabled=no mac-address=$newMac ssid-regexp=\"$newSsid\" place-before=1;"
                :set jobid [:execute script=$asyncCmd]
            }

        } do={
            :set state "Error: something fail on CAPS configuration step: $errorDescr"
            $globalNoteMe value=$state
            :return false
        }

        :return true

    }
}

# Example call
# $globalNewClientCert argClients="anna.ipsec, mikrouter.ipsec" argUsage="tls-client,digital-signature,key-encipherment"
# $globalNewClientCert argClients="anna.capsman, mikrouter.capsman" argUsage="digital-signature,key-encipherment"
# $globalNewClientCert argClients="anna.proxy" argUsage="tls-server,digital-signature,key-encipherment" addSAN="*.anna.home"
# $globalNewClientCert argClients="185.13.148.14" argUsage="tls-server" argBindAsIP="any"
:global globalNewClientCert
:if (!any $globalNewClientCert) do={
    :global globalNewClientCert do={

        # generates IPSEC certs CLIENT TEMPLATE, then requests SCEP to sign it
        # This script is a SCEP-client, it request the server to provide a new certificate
        # it ONLY form the request via API to remote SCEP server

        # incoming named params
        :local clients [:tostr $argClients]
        :local prefs [:tostr $argUsage]
        :local asIp $argBindAsIP
        :local san $addSAN

        # scope global functions
        :global globalNoteMe
        :global globalSecret

        :local state ""

        :if ([:len $clients] > 0) do={
            :if ([:typeof [:tostr $clients]] != "str") do={
                :set state "Error: bad 'clients' parameter passed - ($clients)"
                $globalNoteMe value=$state
                :return false
            }
        } else={
            :set state "Error: bad 'clients' parameter passed - ($clients)"
            $globalNoteMe value=$state
            :return false
        }

        :onerror errorDescr in={

            # clients
            :local ids [:toarray "$clients"]
            :local fakeDomain "myvpn.fake.org"
            :local scepAlias "CHR"
            :set state "Started requests generation"
            $globalNoteMe value=$state

            # this fields should be empty IPSEC/ike2/RSA to work, i can't get it functional with filled fields
            :local country "RU"
            :local certState "MSC"
            :local locality "Moscow"
            :local organization "IKEv2 Home"
            :local unit "IKEv2 Mikrotik"

            :local keySize "2048"

            :local scepUrl "http://185.13.148.14/scep/grant"

            :local tname ""
            :foreach userName in=$ids do={

                :if ([:typeof $san] != "str") do={
                    :set san $userName
                }

                # create a client certificate (that will be just a template while not signed)
                :if ([:len $asIp] > 0) do={

                    :set state "CLIENT TEMPLATE certificates generation as IP...  $userName"
                    $globalNoteMe value=$state

                    :set tname "S.$userName@$scepAlias"

                    :if ([:len [/certificate find where name="$tname"]] > 0) do={
                        :set state "Error: found certificate named ($tname) -  cannot create the same one"
                        $globalNoteMe value=$state
                        :return false
                    } else={
                        /certificate add name="$tname" common-name="$userName@$scepAlias" subject-alt-name="IP:$userName,DNS:$fakeDomain" key-usage=$prefs country="$country" state="$certState" locality="$locality" organization="$organization" unit="$unit" key-size="$keySize" days-valid=365
                    }

                } else={

                    :set state "CLIENT TEMPLATE certificates generation as EMAIL...  $userName"
                    $globalNoteMe value=$state

                    :set tname "C.$userName@$scepAlias"

                    :if ([:len [/certificate find where name="$tname"]] > 0) do={
                        :set state "Error: found certificate named ($tname) -  cannot create the same one"
                        $globalNoteMe value=$state
                        :return false
                    } else={
                        /certificate add name="$tname" common-name="$userName@$scepAlias" subject-alt-name="email:$userName@$fakeDomain,DNS:$san" key-usage=$prefs country="$country" state="$certState" locality="$locality" organization="$organization" unit="$unit" key-size="$keySize" days-valid=365
                    }

                }

                :set state "Pushing sign request..."
                $globalNoteMe value=$state
                /certificate add-scep template="$tname" scep-url="$scepUrl"

                :delay 6s

                # we now have to wait while on remote [mikrotik] this request will be granted and pushed back ready-to-use certificate
                :set state "We now have to wait while on remote [mikrotik] this request will be granted and pushed back ready-to-use certificate... "
                $globalNoteMe value=$state

                :set state "Proceed to remote SCEP please, find this request and approve it. I'll wait 30 seconds"
                $globalNoteMe value=$state

                :delay 30s

                :local baseLength 5
                :for j from=1 to=$baseLength do={
                    :if ([:len [/certificate find where status="idle" name="$tname"]] > 0) do={

                        :set state "Got it at last. Exporting to file"
                        $globalNoteMe value=$state

                        /certificate set trusted=yes [find where name="$tname" and status="idle"]

                        # export the CA, client certificate, and private key
                        # Configure CERTIFICATE_EXPORT_PASSPHRASE in $globalSecret before requesting export.
                        /certificate export-certificate [find where name="$tname" and status="idle"] export-passphrase="$[$globalSecret get CERTIFICATE_EXPORT_PASSPHRASE]" type=pkcs12

                        :return true

                    } else={

                        :set state "Waiting for mikrotik to download the certificate..."
                        $globalNoteMe value=$state
                        :delay 8s

                    }
                }
            }

            :return false

        } do={
            :set state "Error: something fail on SCEP certificates issuing step: $errorDescr"
            $globalNoteMe value=$state
            :return false
        }
    }
}

# Example call
# :put [$globalTimestamp]
# returns "yyyyMMdd-HHmmss" for both old and new RouterOS date formats
:global globalTimestamp
:if (!any $globalTimestamp) do={
    :global globalTimestamp do={
        /system clock
        :local vdate [get date]
        :local vtime [get time]
        :local vdoff [:toarray "0,4,5,7,8,10"]
        :local mon2 [:pick $vdate ($vdoff->2) ($vdoff->3)]
        :local mon [:tonum $mon2]
        :if ($vdate ~ ".../../....") do={
            :set vdoff [:toarray "7,11,1,3,4,6"]
            :set mon ([:find "xxanebarprayunulugepctovecANEBARPRAYUNULUGEPCTOVEC" [:pick $vdate ($vdoff->2) ($vdoff->3)] -1] / 2)
            :if ($mon > 12) do={ :set mon ($mon - 12) }
            :set mon2 [:pick (100 + $mon) 1 3]
        }
        :local yyyy [:pick $vdate ($vdoff->0) ($vdoff->1)]
        :local dd [:pick $vdate ($vdoff->4) ($vdoff->5)]
        :local hh [:pick $vtime 0 2]
        :local mm [:pick $vtime 3 5]
        :local ss [:pick $vtime 6 8]

        :return "$yyyy$mon2$dd-$hh$mm$ss"
    }
}

# Example call
# :local errorDef [$globalCallSftp "RAM/scripts/doBackup.rsc.txt" "REPO/raw/src/scripts/doBackup.rsc.txt"]
# $1=src $2=dst (positional — named src=/dst= empty inside :global do= on field 7.x)
# operation - returns "" on success, error text otherwise
# prefix sftp-session-cut: caller stops the remaining uploads
:global globalCallSftp
:if (!any $globalCallSftp) do={
    :global globalCallSftp do={

        :global globalNoteMe
        :global globalSecret
        :global globalTimestamp

        :local src [:tostr $1]
        :local dst [:tostr $2]

        :if (([:len $src] = 0) or ([:len $dst] = 0)) do={
            :return "sftp: src and dst are required"
        }

        :local ftpServer "$[$globalSecret get SSH_SERVER]"
        :local ftpPort "$[$globalSecret get SSH_PORT]"
        :local ftpUser "$[$globalSecret get SSH_USER]"
        :local ftpPass "$[$globalSecret get SSH_PASSWORD]"
        :local logName ("RAM/" . [$globalTimestamp] . "-" . [:rndstr length=6 from="HtsP2n8qZ"] . ".log.txt")
        # no \" inside "..." — RouterOS ends the string at the inner quote
        :local fetchCmd "/tool fetch url=sftp://$ftpServer:$ftpPort/$dst src-path=$src user=$ftpUser password=$ftpPass upload=yes idle-timeout=30s"

        /file remove [find where name=$logName]
        :local jobId [:execute file=$logName script=$fetchCmd]
        :local ticks 0
        :local protocol ""

        :while (([:len [/system script job find where .id=$jobId]] = 1) && ($ticks < 30)) do={
            :set ticks ($ticks + 1)
            :delay 1s
            :if ([:len [/file find where name=$logName]] = 1) do={
                :set protocol [/file get [find where name=$logName] contents]
            }
            $globalNoteMe value=("sftp " . $src . " to " . $dst . " waiting " . $ticks . " s")
        }

        :if ([:len [/system script job find where .id=$jobId]] = 1) do={
            /system script job remove $jobId
            /file remove [find where name=$logName]
            :return ("sftp-session-cut still running after " . $ticks . " s " . $src . " to " . $dst . " " . $protocol)
        }

        :if ([:len [/file find where name=$logName]] = 1) do={
            :set protocol [/file get [find where name=$logName] contents]
            /file remove [find where name=$logName]
        }

        # missing remote parent dir on canary: uploading uploaded=total then
        # failure: closing connection: <sftp status: 20> … (/tool/fetch; line 1)
        :local dirHint " hint: check remote parent dir exists (fetch does not mkdir); or SSH ban"

        :if (($protocol ~ "status: failed") || ($protocol ~ "sftp status: 20") || ($protocol ~ "timed out") || ($protocol ~ "timeout")) do={
            :return ("sftp-session-cut " . $src . " to " . $dst . " " . $protocol . $dirHint)
        }

        :if ($protocol ~ "status: finished") do={
            $globalNoteMe value=("sftp finished " . $src . " to " . $dst)
            :return ""
        }

        :return ("sftp upload failed " . $src . " to " . $dst . " " . $protocol . $dirHint)
    }
}

        :onerror errorDescr in={

            :set state "Call: /system ssh-exec address=$sshServer user=$sshUser port=$sshPort command=\"$asyncCmd\" as-value"
            $globalNoteMe value=$state

            :local result [/system ssh-exec address=$sshServer user=$sshUser password=$sshPass port=$sshPort command=$asyncCmd as-value]
            :local exitCode ([$result]->"exit-code")

            :if ($exitCode != 0) do={
                :set state "RPC: ssh command returned exit code ($exitCode)"
            } else={
                # no errors - success operation
                :set state ""
            }

        } do={
            :set state "RPC: ssh error when $state: $errorDescr"
            $globalNoteMe value=$state
        }

        :return $state
    }
}
