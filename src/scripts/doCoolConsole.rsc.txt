:local sysname [/system identity get name]
:local scriptname "doCoolConsole"
:global globalScriptBeforeRun
$globalScriptBeforeRun $scriptname

:global globalNoteMe
:global globalTgMessage
:global globalOnPrimaryPartition
:local itsOk true
:local state ""

:local logcontenttemp ""
:local logcontent ""

# predicate-like helper: returns resolved address or "ERROR"
:local safeResolve do={
    :local host "ERROR"
    :if ([:len $1] != 0) do={
        :onerror errorDescr in={
            :set host [:resolve "$1"]
        } do={
            :log error "FAIL resolving: $1 ($errorDescr)"
            :put "FAIL resolving: $1"
        }
    } else={
        :log error "FAIL resolving: empty host name"
        :put "FAIL resolving: empty host name"
    }
    :return $host
}

:onerror errorDescr in={

    :local rosVer [:tonum [:pick [/system resource get version] 0 1]]
    :local sysver "NA"
    :if ([:len [/system package find where name="system" and disabled=no]] > 0 and $rosVer = 6) do={
        :set sysver [/system package get system version]
    }
    :if ([:len [/system package find where name="routeros" and disabled=no]] > 0 and $rosVer = 7) do={
        :set sysver [/system package get routeros version]
    }

    :set state "Picking default route"

    :local defaultRoute "unreachable"
    /ip route {
        :foreach i in=[find where dst-address="0.0.0.0/0" and active and routing-table=main] do={
            :set defaultRoute [:tostr [/ip route get $i gateway]]
        }
    }

    :set state "Picking ipsec"

    :local ipsecState "okay"
    /ip ipsec policy {
        :foreach vpnEndpoint in=[find (!disabled and !template)] do={
            :local ph2state [get value-name=ph2-state $vpnEndpoint]
            :if ($ph2state != "established") do={
                :set ipsecState "issues found"
            }
        }
    }

    :if (![$globalOnPrimaryPartition]) do={
        :set state "WARNING: the system booted up from fallback partition!"
        :log error $state
        $globalNoteMe value=$state

        :set logcontenttemp "$state"
        :set logcontent ("$logcontent" . "$logcontenttemp" . " \n")
    }

    :set logcontenttemp "Ipsec:         $ipsecState"
    :set logcontent ("$logcontent" . "$logcontenttemp" . " \n")
    :set logcontenttemp "Route:     $defaultRoute"
    :set logcontent ("$logcontent" . "$logcontenttemp" . " \n")
    :set logcontenttemp "Version:         $sysver"
    :set logcontent ("$logcontent" . "$logcontenttemp" . " \n")
    :set logcontenttemp "Uptime:        $[/system resource get uptime]"
    :set logcontent ("$logcontent" . "$logcontenttemp" . "  \n")

    :local avgRttA 0
    :local numPing 6
    :local latency "NA"

    :set state "Yandex connection test"

    :local latencySite "Ya.ru"
    :local yaResolve [$safeResolve $latencySite]

    :set state "Picking latency"

    :if ($yaResolve != "ERROR") do={
        :for tmpA from=1 to=$numPing step=1 do={
            /tool flood-ping count=1 size=38 address=$yaResolve do={ :set avgRttA ($"avg-rtt" + $avgRttA) }
            :delay 1s
        }
        :set latency [:tostr ($avgRttA / $numPing)]
    } else={
        :set latency "unreachable"
    }

    :set state "Resolving CHR, MIK, ANNA"

    :local chrResolve [$safeResolve "accb195e0dffc6bb.sn.mynetname.net"]
    :if ($chrResolve = "ERROR") do={ :set chrResolve "unreachable" }

    :local mikResolve [$safeResolve "673706ed7949.sn.mynetname.net"]
    :if ($mikResolve = "ERROR") do={ :set mikResolve "unreachable" }

    :local annaResolve [$safeResolve "hcy086pz6xz.sn.mynetname.net"]
    :if ($annaResolve = "ERROR") do={ :set annaResolve "unreachable" }

    :set logcontenttemp "Time:        $[/system clock get date] $[/system clock get time]"
    :set logcontent ("$logcontent" . "$logcontenttemp" . "  \n")
    :set logcontenttemp "Ping:    $latency ms"
    :set logcontent ("$logcontent" . "$logcontenttemp" . "  \n")

    :set logcontenttemp "Chr:        $chrResolve"
    :set logcontent ("$logcontent" . "$logcontenttemp" . "  \n")
    :set logcontenttemp "Mik:        $mikResolve"
    :set logcontent ("$logcontent" . "$logcontenttemp" . "  \n")
    :set logcontenttemp "Anna:        $annaResolve"
    :set logcontent ("$logcontent" . "$logcontenttemp" . "  \n")

    :set logcontenttemp "Clock:        $[/system ntp client get status]"
    :set logcontent ("$logcontent" . "$logcontenttemp" . "  \n")

    :set state "Listing packages"

    /system/package {
        :foreach pkg in=[find (!disabled)] do={
            :local pkgName [get value-name=name $pkg]
            :set logcontent ("$logcontent" . " * $pkgName" . "  \n")
        }
    }

    :set state "Setting global note"
    $globalNoteMe value=$state

    /system note set note="$logcontent"
    $globalNoteMe value="Done"

} do={
    :set state "Error When $state: $errorDescr"
    $globalNoteMe value=$state
    :set itsOk false
}

:local inf ""
:if ($itsOk) do={
    :set inf "✓ SUCCESS $scriptname on $sysname: console note refreshed"
}

:if (!$itsOk) do={
    :set inf "✗ FAILED $scriptname on $sysname: $state"
}

$globalNoteMe value=$inf

:if (!$itsOk) do={
    $globalTgMessage value=$inf
    :error $inf
}
