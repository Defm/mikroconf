:local sysname [/system identity get name]
:local scriptname "doStartupScript"

# reset current
/system note set note="Pending"

# Track sync time (avoid CHR initial clock bug), with this we are also checking if internet comes up
# globalNoteMe does not exist yet - direct :log/:put here
:onerror errorDescr in={

    :local successSyncState "synchronized"
    :local syncOk false
    :local syncState ""
    :local timeSpent 0
    :local ticks 0
    :local maxTicks 40
    :local break false
    :do {
        :set syncState [/system ntp client get status]
        :set syncOk ($syncState = $successSyncState)

        :log info "Waiting 15s for clock sync using NTP.. ($ticks/$maxTicks, $syncState)"
        :put "Waiting 15s for clock sync using NTP.. ($ticks/$maxTicks, $syncState)"

        :if (!$syncOk) do={
            :delay 15s
            :set timeSpent ($timeSpent + 15)
            :set break ($ticks >= $maxTicks)
        } else={
            :set break true
        }

        :set ticks ($ticks + 1)

    } while=(!$break)

    :if ($syncOk) do={
        :log warning "Successful clock sync using NTP in $timeSpent seconds"
        :put "Successful clock sync using NTP in $timeSpent seconds"
    } else={
        :log error "Error when clock sync using NTP in $timeSpent seconds"
        :put "Error when clock sync using NTP in $timeSpent seconds"
    }

} do={
    :log error "Error when tracking clock sync using NTP: $errorDescr"
    :put "Error when tracking clock sync using NTP: $errorDescr"
}

# bootstrap environment directly - globalSafeScriptCall appears only after doEnvironmentSetup
:onerror errorDescr in={
    :log warning "Starting script: doEnvironmentClearance (STARTUP context)"
    /system script run doEnvironmentClearance
    :log warning "Starting script: doEnvironmentSetup (STARTUP context)"
    /system script run doEnvironmentSetup
} do={
    :log error "FAIL Starting environment scripts (STARTUP context): $errorDescr"
    :put "FAIL Starting environment scripts (STARTUP context): $errorDescr"
}

:global globalSafeScriptCall
:global globalNoteMe
:global globalTgMessage

:if (any $globalSafeScriptCall) do={
    $globalSafeScriptCall "doImperialMarch" "STARTUP"
    $globalSafeScriptCall "doCoolConsole" "STARTUP"
} else={
    :log error "Environment is not ready - skipping startup chain (STARTUP context)"
}

# wait some for all tunnels to come up after reboot and VPN to work
:local inf "Wait some for all tunnels to come up after reboot and VPN to work.."
:if (any $globalNoteMe) do={ $globalNoteMe value=$inf }

:delay 15s

:set inf "$scriptname on $sysname: system restart detected"
:if (any $globalNoteMe) do={ $globalNoteMe value=$inf }
:if (any $globalTgMessage) do={ $globalTgMessage value=$inf }
