# NetWatch notifier OnUp/OnDown - no external globals, priorities are configured below

# init globals as far as we are inside *sys user account
:global globalSafeScriptCall
:if (!any $globalSafeScriptCall) do={ /system script run doEnvironmentSetup }

:local sysname [/system identity get name]
:local scriptname "doNetwatchDNS"
:global globalScriptBeforeRun
$globalScriptBeforeRun $scriptname "NETWATCH"

:global globalNoteMe
:global globalTgMessage
:local itsOk true
:local state ""

# In case when the host has two or more (eth+wlan+lte etc) interfaces connected to our network
# - we need to ensure we have the only one DNS entry points to IP that is online (and track disconnections)
:local priorities [{}]
:set ($priorities->"wb.home") {"192.168.90.2"; "192.168.90.3"}

# placeholder address while checkup is running (so we get Dns-NxDomain error when resolve over Chrome etc)
:local dummyIp "172.16.0.10"

:foreach dnsName,ipAddrs in=$priorities do={

    :onerror errorDescr in={

        :local currentIp
        :local dnsEntry ([/ip/dns/static/find name=$dnsName]->0)

        :set state "Netwatch for $dnsName started..."
        $globalNoteMe value=$state

        :local stamp [/system clock get time]

        :if ([:typeof $dnsEntry] = "id") do={
            :set currentIp [/ip/dns/static get $dnsEntry address]
            # Flush entry address in any case and set Disabled
            /ip dns static set $dnsEntry address=$dummyIp disabled=yes comment="Netwatch checkup is running..."
        } else={
            :set currentIp $dummyIp
            /ip dns static add name=$dnsName address=$currentIp disabled=yes comment="Netwatch checkup is running..."
            :set dnsEntry ([/ip/dns/static/find name=$dnsName]->0)
        }
        :set state "Current DNS entry for $dnsName: $currentIp"
        $globalNoteMe value=$state

        :local targetIp "0.0.0.1"

        # "break-loop" leaves the priorities loop as soon as the first online IP is found
        :onerror errorDescr in={

            :foreach ipAddr in=$ipAddrs do={

                :local addr [:toip $ipAddr]

                :set state "Testing if $ipAddr connected.."
                $globalNoteMe value=$state

                :local received [/ping address="$addr" count=5 interval=1000ms]

                :if ($received > 3) do={

                    :if ($ipAddr != $currentIp) do={
                        /ip dns static set $dnsEntry disabled=no address=$ipAddr comment="Netwatch checkup at $stamp"
                        :set state "DNS updated to $ipAddr ($dnsName)"
                        $globalNoteMe value=$state
                        /ip/dns/cache/flush
                    } else={
                        /ip dns static set $dnsEntry disabled=no address=$ipAddr comment="Netwatch checkup at $stamp"
                        :set state "DNS $dnsName already points to $ipAddr"
                        $globalNoteMe value=$state
                    }

                    :set targetIp $addr

                    # no need to process less priority IPs
                    :error "break-loop"

                } else={
                    :set state "It looks like $ipAddr is offline, skipping it"
                    $globalNoteMe value=$state
                }
            }

        } do={
            :if (!($errorDescr ~ "^break-loop")) do={
                :error $errorDescr
            }
        }

        :set state "Finished processing $dnsName with target ip $targetIp"
        $globalNoteMe value=$state

    } do={
        :set state "Error When $state: $errorDescr"
        $globalNoteMe value=$state
        :set itsOk false
    }
}

:local inf ""
:if ($itsOk) do={
    :set inf "✓ SUCCESS $scriptname on $sysname: DNS entries refreshed"
}

:if (!$itsOk) do={
    :set inf "✗ FAILED $scriptname on $sysname: $state"
}

$globalNoteMe value=$inf

:if (!$itsOk) do={
    $globalTgMessage value=$inf
    :error $inf
}
