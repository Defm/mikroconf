:local sysname [/system identity get name]
:local scriptname "doCloudBackup"
:global globalScriptBeforeRun
$globalScriptBeforeRun $scriptname

:global globalNoteMe
:global globalTgMessage
:global globalSecret
:global globalOnPrimaryPartition
:local itsOk true
:local state ""

# Configure at runtime: $globalSecret set BACKUP_PASSWORD password="YOUR_SECRET"
:local backupPassword "$[$globalSecret get BACKUP_PASSWORD]"

# preflight
:if (![$globalOnPrimaryPartition]) do={
    :set state "WARNING: the system booted up from fallback partition - skipping backup!"
    :log error $state
    $globalNoteMe value=$state
    :error $state
}

:onerror errorDescr in={

    # we are not interested in output, but print without count-only is
    # required to fetch information from cloud
    /system backup cloud print as-value

    :set state "Creating and uploading cloud backup"
    $globalNoteMe value=$state

    :local backup ([/system/backup/cloud/find]->0)
    :if ([:typeof $backup] = "id") do={
        /system/backup/cloud/upload-file action=create-and-upload password=$backupPassword replace=$backup
    } else={
        /system/backup/cloud/upload-file action=create-and-upload password=$backupPassword
    }

    :set backup ([/system/backup/cloud/find]->0)
    :local backupName [/system/backup/cloud/get $backup name]
    :set state "Cloud backup file uploaded: $backupName"
    $globalNoteMe value=$state
    $globalNoteMe value="Done"

} do={
    :set state "Error When $state: $errorDescr"
    $globalNoteMe value=$state
    :set itsOk false
}

:local inf ""
:if ($itsOk) do={
    :set inf "✓ SUCCESS $scriptname on $sysname: cloud backup uploaded"
}

:if (!$itsOk) do={
    :set inf "✗ FAILED $scriptname on $sysname: $state"
}

$globalNoteMe value=$inf

:if (!$itsOk) do={
    $globalTgMessage value=$inf
    :error $inf
}
