:local sysname [/system identity get name]
:local scriptname "doFreshTheScripts"
:global globalScriptBeforeRun
$globalScriptBeforeRun $scriptname

:global globalNoteMe
:global globalTgMessage
:local itsOk true
:local state ""

:local gitHubUserName "Defm"
:local gitHubRepoName "mikroconf"

# should be used for private repos
:local gitHubAccessToken ""

:local requestUrl "https://$gitHubAccessToken@relay.usetheforce.io/$gitHubUserName/$gitHubRepoName/master/src/scripts/"

# whitelist: scripts absent on the device are skipped with a message
:local useUpdateList true
:local updateList [:toarray "doBackup,doEnvironmentSetup,doEnvironmentClearance,doRandomGen,doFreshTheScripts,doCertificatesIssuing,doSCEPClientCertificatesIssuing,doNetwatchHost,doNetwatchDNS,doDHCPLeaseTrack,doIPSECPunch,doStartupScript,doHeatFlag,doPeriodicLogDump,doPeriodicLogParse,doTelegramNotify,doLEDoff,doLEDon,doCPUHighLoadReboot,doUpdatePoliciesRemotely,doUpdateExternalDNS,doSuperviseCHRviaSSH,doCoolConsole,doFlushLogs,doCloudBackup,doDumpTheScripts,doKeepScriptsOwner,doFreshDNSAddressLists,doPushStatsToInfluxDB,doCreateTrafficAccountingQueues,doCheckPingRate,doStaleTSLConnectionsTrack,doCollectSpeedStats,doWestminister,doBumerSound,doSomeAlarm,doImperialMarch,flush_dns"]

:foreach scriptName in=$updateList do={
    :if ([:len [/system script find name=$scriptName]] = 0) do={
        :set state "Script '$scriptName' skipped due to absence"
        $globalNoteMe value=$state
    }
}

:foreach scriptId in=[/system script find] do={

    :local theScript [/system script get $scriptId name]

    :onerror errorDescr in={

        :if ($useUpdateList and [:len [find key=$theScript in=$updateList]] = 0) do={
            :set state "Script '$theScript' skipped due to setup"
            $globalNoteMe value=$state
            :error "skip-continue"
        }

        :set state "/tool fetch url=$requestUrl$$theScript.rsc.txt output=user as-value"
        $globalNoteMe value=$state

        # Please keep care about consistency if size over 4096 bytes
        :local answer [/tool fetch url="$requestUrl$$theScript.rsc.txt" output=user as-value]
        :local code ($answer->"data")
        $globalNoteMe value="Done"

        :set state "Setting Up Script source for '$theScript'"
        $globalNoteMe value=$state
        /system script set $theScript source="$code"
        $globalNoteMe value="Done"

        :delay 1s

    } do={
        # scheduler may append location to :error text; match marker as prefix
        :if (!($errorDescr ~ "^skip-continue")) do={
            :set state "Error When $state: $errorDescr"
            $globalNoteMe value=$state
            :set itsOk false
        }
    }
}

:local inf ""
:if ($itsOk) do={
    :set inf "✓ SUCCESS $scriptname on $sysname: scripts refreshed"
}

:if (!$itsOk) do={
    :set inf "✗ FAILED $scriptname on $sysname: $state"
}

$globalNoteMe value=$inf

:if (!$itsOk) do={
    $globalTgMessage value=$inf
    :error $inf
}
