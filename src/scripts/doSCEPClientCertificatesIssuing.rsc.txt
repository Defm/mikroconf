:local sysname [/system identity get name]
:local scriptname "doSCEPClientCertificatesIssuing"
:global globalScriptBeforeRun
$globalScriptBeforeRun $scriptname

:global globalNoteMe
:global globalTgMessage
:local itsOk true
:local state ""

# generates IPSEC certs CLIENT TEMPLATE, then requests SCEP to sign it

# clients
:local ids [:toarray "alx.iphone.rw.2021"]
:local fakeDomain "myvpn.fake.org"

:local scepAlias "CHR"
:local scepUrl "http://185.13.148.14/scep/grant"

# subject fields should be empty for IPSEC/ike2/RSA to work (filled fields were not functional)
:local country ""
:local certState ""
:local locality ""
:local organization ""
:local unit ""

:local keySize "2048"

:onerror errorDescr in={

    :foreach userName in=$ids do={

        :set state "CLIENT TEMPLATE certificates generation...  $userName"
        $globalNoteMe value=$state

        # create a client certificate (that will be just a template while not signed)
        /certificate add name="$userName@$scepAlias" common-name="$userName@$scepAlias" subject-alt-name="email:$userName@$fakeDomain" key-usage=tls-client country="$country" state="$certState" locality="$locality" organization="$organization" unit="$unit" key-size="$keySize" days-valid=365

        :set state "Pushing sign request for $userName..."
        $globalNoteMe value=$state

        /certificate add-scep template="$userName@$scepAlias" scep-url="$scepUrl"

        :delay 6s

        # we now have to wait while on remote [mikrotik] this request will be granted and pushed back ready-to-use certificate
        :set state "We now have to wait while on remote [mikrotik] this request will be granted and pushed back ready-to-use certificate... Proceed to remote SCEP please"
        $globalNoteMe value=$state

    }

    $globalNoteMe value="Done"

} do={
    :set state "Error When $state: $errorDescr"
    $globalNoteMe value=$state
    :set itsOk false
}

:local inf ""
:if ($itsOk) do={
    :set inf "✓ SUCCESS $scriptname on $sysname: SCEP requests pushed"
}

:if (!$itsOk) do={
    :set inf "✗ FAILED $scriptname on $sysname: $state"
}

$globalNoteMe value=$inf

:if (!$itsOk) do={
    $globalTgMessage value=$inf
    :error $inf
}
