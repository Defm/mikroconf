:local sysname [/system identity get name]
:local scriptname "doAdblock"
:global globalScriptBeforeRun
$globalScriptBeforeRun $scriptname

:global globalNoteMe
:global globalTgMessage
:local itsOk true
:local state ""

# StopAD hosts-based ad block (vendor logic, wrapped in author form)
# https://github.com/tarampampam/mikrotik-hosts-parser
:local hostScriptUrl "https://stopad.cgood.ru/script/source?format=routeros&version=2.0.2&sources_urls=https%3A%2F%2Fcdn.rawgit.com%2Ftarampampam%2Fstatic%2Fmaster%2Fhosts%2Fblock_shit.txt,http%3A%2F%2Fadaway.org%2Fhosts.txt,http%3A%2F%2Fpgl.yoyo.org%2Fadservers%2Fserverlist.php%3Fhostformat%3Dhosts%26showintro%3D0%26mimetype%3Dplaintext&excluded_hosts=localhost"
:local stopAdScriptName "stop_ad.script"
:local backupFileName "before_stopad"

:onerror errorDescr in={

    :set state "Fetching StopAD script"
    $globalNoteMe value=$state
    /tool fetch check-certificate=no mode=https url=$hostScriptUrl dst-path=("./" . $stopAdScriptName)

    :if ([:len [/file find name=$stopAdScriptName]] > 0) do={

        :set state "Saving backup before StopAD import"
        $globalNoteMe value=$state
        /system backup save name=$backupFileName
        :delay 1s

        :if ([:len [/file find name=($backupFileName . ".backup")]] > 0) do={
            :set state "Importing StopAD static DNS entries"
            $globalNoteMe value=$state
            /ip dns static remove [/ip dns static find comment=ADBlock]
            /import file-name=$stopAdScriptName
            /file remove $stopAdScriptName
            $globalNoteMe value="Done"
        } else={
            :set state "Backup file not created, importing AD block script stopped"
            $globalNoteMe value=$state
            :set itsOk false
        }

    } else={
        :set state "AD block script not downloaded, script stopped"
        $globalNoteMe value=$state
        :set itsOk false
    }

} do={
    :set state "Error When $state: $errorDescr"
    $globalNoteMe value=$state
    :set itsOk false
}

:local inf ""
:if ($itsOk) do={
    :set inf "✓ SUCCESS $scriptname on $sysname: StopAD imported"
}

:if (!$itsOk) do={
    :set inf "✗ FAILED $scriptname on $sysname: $state"
}

$globalNoteMe value=$inf

:if (!$itsOk) do={
    $globalTgMessage value=$inf
    :error $inf
}
