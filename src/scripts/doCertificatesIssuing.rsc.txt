:local sysname [/system identity get name]
:local scriptname "doCertificatesIssuing"
:global globalScriptBeforeRun
$globalScriptBeforeRun $scriptname

:global globalNoteMe
:global globalTgMessage
:global globalSecret
:local itsOk true
:local state ""

# generates IPSEC certs: CA, server, IOS *.mobileconfig profile sign and clients
# i recommend to run it on server side

# clients
:local ids [:toarray "alx.iphone.rw.20-21,alx.mbp.rw.20-21"]
:local fakeDomain "myvpn.fake.org"

# Configure CERTIFICATE_EXPORT_PASSPHRASE in $globalSecret before running.
:local certificateExportPassphrase "$[$globalSecret get CERTIFICATE_EXPORT_PASSPHRASE]"

# subject fields should be empty for IPSEC/ike2/RSA to work (filled fields were not functional)
:local country ""
:local certState ""
:local locality ""
:local organization ""
:local unit ""

:local keySize "2048"

:local maskedServerIp [/ip address get [find where interface=wan] address]
:local serverIp [:pick "$maskedServerIp" 0 [:find "$maskedServerIp" "/" -1]]

:onerror errorDescr in={

    :set state "CA certificates generation..."
    $globalNoteMe value=$state

    # generate a CA certificate (that will be just a template while not signed)
    # crl-sign allows to use SCEP
    /certificate add name="ca.$fakeDomain" common-name="ca@$sysname" subject-alt-name="DNS:ca.$fakeDomain" key-usage=crl-sign,key-cert-sign country="$country" state="$certState" locality="$locality" organization="$organization" unit="$unit" key-size="$keySize" days-valid=3650

    :set state "Signing CA..."
    $globalNoteMe value=$state

    /certificate sign "ca.$fakeDomain" ca-crl-host="$serverIp" name="ca@$sysname"

    :delay 6s

    /certificate set trusted=yes "ca@$sysname"

    :set state "Exporting CA as PEM..."
    $globalNoteMe value=$state

    # export the CA, as PEM
    /certificate export-certificate "ca@$sysname" type=pem

    :set state "SERVER certificates generation..."
    $globalNoteMe value=$state

    # generate a server certificate (that will be just a template while not signed)
    /certificate add name="server.$fakeDomain" common-name="server@$sysname" subject-alt-name="IP:$serverIp,DNS:$fakeDomain" key-usage=tls-server country="$country" state="$certState" locality="$locality" organization="$organization" unit="$unit" key-size="$keySize" days-valid=365

    :set state "Signing server..."
    $globalNoteMe value=$state

    /certificate sign "server.$fakeDomain" ca="ca@$sysname" name="server@$sysname"

    :delay 6s

    /certificate set trusted=yes "server@$sysname"

    :set state "CODE SIGN certificates generation..."
    $globalNoteMe value=$state

    # generate a code signing (apple IOS profiles) certificate (that will be just a template while not signed)
    /certificate add name="sign.$fakeDomain" common-name="sign@$sysname" subject-alt-name="DNS:sign.$fakeDomain" key-usage=code-sign,digital-signature country="$country" state="$certState" locality="$locality" organization="$organization" unit="$unit" key-size="$keySize" days-valid=365

    :set state "Signing code sign..."
    $globalNoteMe value=$state

    /certificate sign "sign.$fakeDomain" ca="ca@$sysname" name="sign@$sysname"

    :delay 6s

    /certificate set trusted=yes "sign@$sysname"

    # export the CA, code sign certificate, and private key
    /certificate export-certificate "sign@$sysname" export-passphrase=$certificateExportPassphrase type=pkcs12

    :foreach userName in=$ids do={

        :set state "CLIENT certificates generation...  $userName"
        $globalNoteMe value=$state

        # create a client certificate (that will be just a template while not signed)
        /certificate add name="client.$fakeDomain" common-name="$userName@$sysname" subject-alt-name="email:$userName@$fakeDomain" key-usage=tls-client country="$country" state="$certState" locality="$locality" organization="$organization" unit="$unit" key-size="$keySize" days-valid=365

        :set state "Signing client $userName..."
        $globalNoteMe value=$state

        /certificate sign "client.$fakeDomain" ca="ca@$sysname" name="$userName@$sysname"

        :delay 6s

        /certificate set trusted=yes "$userName@$sysname"

        # export the CA, client certificate, and private key
        /certificate export-certificate "$userName@$sysname" export-passphrase=$certificateExportPassphrase type=pkcs12

    }

    $globalNoteMe value="Done"

} do={
    :set state "Error When $state: $errorDescr"
    $globalNoteMe value=$state
    :set itsOk false
}

:local inf ""
:if ($itsOk) do={
    :set inf "✓ SUCCESS $scriptname on $sysname: certificates issued and exported"
}

:if (!$itsOk) do={
    :set inf "✗ FAILED $scriptname on $sysname: $state"
}

$globalNoteMe value=$inf

:if (!$itsOk) do={
    $globalTgMessage value=$inf
    :error $inf
}
