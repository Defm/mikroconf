# fill it inside dhcp lease-script
:global globalLeaseBound
:global globalLeaseServerName
:global globalLeaseActMac
:global globalLeaseActIp

# may be triggered twice per moment - snapshot globals first
:local leaseBound $globalLeaseBound
:local leaseServerName $globalLeaseServerName
:local leaseActMac $globalLeaseActMac
:local leaseActIp $globalLeaseActIp

# init globals as far as we are inside *sys user account
:global globalSafeScriptCall
:if (!any $globalSafeScriptCall) do={ /system script run doEnvironmentSetup }

:local sysname [/system identity get name]
:local scriptname "doDHCPLeaseTrack"
:global globalScriptBeforeRun
$globalScriptBeforeRun $scriptname "DHCP"

:global globalNoteMe
:global globalTgMessage
:local itsOk true
:local state ""

:if (!any $leaseActIp) do={
    :set state "No globalLeaseActIp provided.."
    $globalNoteMe value=$state
    :error $state
}

:if (!any $leaseServerName) do={
    :set state "No globalLeaseServerName provided.."
    $globalNoteMe value=$state
    :error $state
}

:onerror errorDescr in={

    :if ($leaseBound = 1) do={

        :local leaseId [/ip dhcp-server lease find where server=$leaseServerName and address=$leaseActIp]

        :if ([:len $leaseId] = 0) do={
            :set state "DHCP lease not found for server=$leaseServerName ip=$leaseActIp"
            $globalNoteMe value=$state
        } else={
            :local clientid [/ip dhcp-server lease get $leaseId host-name]
            :set state "DHCP Client assigned - $clientid ($leaseActIp)"
            $globalNoteMe value=$state
            $globalTgMessage value=$state
        }

    } else={
        :set state "DHCP Client expired - $leaseActIp"
        $globalNoteMe value=$state
    }

} do={
    :set state "Error When $state: $errorDescr"
    $globalNoteMe value=$state
    :set itsOk false
}

:local inf ""
:if ($itsOk) do={
    :set inf "✓ SUCCESS $scriptname on $sysname: $state"
}

:if (!$itsOk) do={
    :set inf "✗ FAILED $scriptname on $sysname: $state"
}

$globalNoteMe value=$inf

:if (!$itsOk) do={
    $globalTgMessage value=$inf
    :error $inf
}
