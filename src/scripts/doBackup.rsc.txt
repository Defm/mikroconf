:local sysname [/system identity get name]
:local scriptname "doBackup"
:global globalScriptBeforeRun
$globalScriptBeforeRun $scriptname

:global globalNoteMe
:global globalTgMessage
:global globalSecret
:global globalTimestamp
:global globalCallSftp
:global globalKnockKnock
:global globalOnPrimaryPartition
:local itsOk true
:local state ""
:local firstError ""

:local rosVer [:tonum [:pick [/system resource get version] 0 1]]
:local sysver "NA"
:if ([:len [/system package find where name="system" and disabled=no]] > 0 and $rosVer = 6) do={
    :set sysver [/system package get system version]
}
:if ([:len [/system package find where name="routeros" and disabled=no]] > 0 and $rosVer = 7) do={
    :set sysver [/system package get routeros version]
}

:local useSysBackup true
:local useEncryption false
:local useRawExport true

# paths — host /REPO/raw is the git clone (same as doDumpTheScripts)
:local useFtp true
:local ftpServer "$[$globalSecret get SSH_SERVER]"
:local ftpBinRoot "REPO/raw/bin/"
:local useFtpGit true
:local ftpConfigLatest ("REPO/raw/src/config/" . $sysname . ".safe.rsc")
:local configLatestName ($sysname . ".safe.rsc")
:local ramDiskName "RAM"
:local ramDiskPrefix ($ramDiskName . "/")
:local ramBackupPattern ("^" . $ramDiskPrefix . "BACKUP-" . $sysname . "-")
:local ramConfigLatest ($ramDiskPrefix . $configLatestName)

:local sysnote [/system note get note]

:local stamp [$globalTimestamp]

:local useSmtp true
:local smtpAddress "defm.kopcap@gmail.com"
:local smtpSubject "$sysname Full Backup ($stamp)"
:local smtpBody "$sysname full Backup file see in attachment.\n $sysnote"

# preflight
:if (![$globalOnPrimaryPartition]) do={
    :set state "WARNING: the system booted up from fallback partition - skipping backup!"
    :log error $state
    $globalNoteMe value=$state
    :error $state
}

:if ([:len [/disk find where slot=$ramDiskName and type="tmpfs"]] = 0) do={
    :set state "WARNING: tmpfs '$ramDiskName' not found - skipping backup!"
    :log error $state
    $globalNoteMe value=$state
    :error $state
}

:onerror errorDescr in={
    :local smtpserv [:resolve "$ftpServer"]
} do={
    :set state "WARNING: FTP server '$ftpServer' unreachable - skipping backup!"
    :log error $state
    $globalNoteMe value=$state
    :error $state
}

:local fname "BACKUP-$sysname-$stamp"

$globalNoteMe value="Clearing previous backups.." verbose=true
/file remove [/file find where name~$ramBackupPattern]
/file remove [/file find where name=$ramConfigLatest]

:onerror errorDescr in={

    :if ($useSysBackup) do={
        :set state "Saving system backup $fname.backup"
        $globalNoteMe value=$state verbose=true
        :if ($useEncryption) do={
            # Configure at runtime: $globalSecret set BACKUP_PASSWORD password="YOUR_SECRET"
            :local backupPassword "$[$globalSecret get BACKUP_PASSWORD]"
            /system backup save name=($ramDiskName . "/" . $fname . ".backup") password=$backupPassword
        }
        :if (!$useEncryption) do={ /system backup save dont-encrypt=yes name=($ramDiskName . "/" . $fname . ".backup") }
        :delay 2s
        $globalNoteMe value="System Backup Finished" verbose=true
    }

    :if ($useRawExport) do={
        :set state "Exporting restore text $fname.unsafe.rsc"
        $globalNoteMe value=$state verbose=true
        /export show-sensitive terse file=($ramDiskName . "/" . $fname . ".unsafe.rsc")
        :delay 2s
        $globalNoteMe value="Done" verbose=true
    }

    :if ($useFtpGit) do={
        :set state "Exporting config source $configLatestName"
        $globalNoteMe value=$state verbose=true
        /export terse file=($ramDiskName . "/" . $configLatestName)
        :delay 2s
        $globalNoteMe value="Done" verbose=true
    }

    :delay 5s

} do={
    :set state "Error When $state: $errorDescr"
    $globalNoteMe value=$state
    :if ([:len $firstError] = 0) do={ :set firstError $state }
    :set itsOk false
}

:if ($itsOk) do={

    # "break-loop" leaves remaining uploads after sftp-session-cut
    :onerror errorDescr in={

        :if ($useFtp) do={
            # SFTP uploads look like bruteforce on CHR without knockknock
            :set state "Opening firewall for SFTP (knockknock)"
            $globalNoteMe value=$state verbose=true
            :local errorDef [$globalKnockKnock $scriptname]
            :if ([:len $errorDef] > 0) do={ :error $errorDef }
        }

        :foreach backupFile in=[/file find] do={

            :local buFile [/file get $backupFile name]

            :if ([:typeof [:find $buFile $fname]] != "nil") do={

                :local itsUnsafe ($buFile ~ ".unsafe.rsc")

                :if ($useFtp) do={

                    :onerror errorDescr in={
                        :local dst "$ftpBinRoot$buFile"
                        :if ([:typeof [:find $buFile $ramDiskPrefix]] = "num") do={
                            :set dst ("$ftpBinRoot" . [:pick $buFile [:len $ramDiskPrefix] [:len $buFile]])
                        }
                        :set state "Uploading $buFile to SFTP ($dst)"
                        $globalNoteMe value=$state verbose=true

                        :local errorDef [$globalCallSftp $buFile $dst]
                        :if ([:len $errorDef] > 0) do={ :error $errorDef }

                        $globalNoteMe value="Done" verbose=true
                    } do={
                        :set state "Error When $state: $errorDescr"
                        $globalNoteMe value=$state
                        :if ([:len $firstError] = 0) do={ :set firstError $state }
                        :set itsOk false
                        :if ($errorDescr ~ "^sftp-session-cut") do={
                            :error "break-loop"
                        }
                    }
                }

                :if ($useSmtp and !$itsUnsafe) do={
                    :onerror errorDescr in={
                        :set state "Uploading $buFile to SMTP"
                        $globalNoteMe value=$state verbose=true

                        # email works in background, delay needed
                        /tool e-mail send to=$smtpAddress body=$smtpBody subject=$smtpSubject file=$buFile tls=starttls

                        # waiting for email to be delivered
                        :delay 15s

                        :if ([/tool e-mail get last-status] != "succeeded") do={
                            :error "e-mail last-status is not succeeded"
                        }

                        $globalNoteMe value="Done" verbose=true
                    } do={
                        :set state "Error When $state: $errorDescr"
                        $globalNoteMe value=$state
                        :if ([:len $firstError] = 0) do={ :set firstError $state }
                        :set itsOk false
                    }
                }
            }
        }

        :if ($useFtp and $useFtpGit) do={
            :onerror errorDescr in={
                :if ([:len [/file find name=$ramConfigLatest]] = 0) do={
                    :error "config source $configLatestName not found"
                }
                :set state "Uploading $configLatestName to SFTP ($ftpConfigLatest)"
                $globalNoteMe value=$state verbose=true
                :local errorDef [$globalCallSftp $ramConfigLatest $ftpConfigLatest]
                :if ([:len $errorDef] > 0) do={ :error $errorDef }
                $globalNoteMe value="Done" verbose=true
            } do={
                :set state "Error When $state: $errorDescr"
                $globalNoteMe value=$state
                :if ([:len $firstError] = 0) do={ :set firstError $state }
                :set itsOk false
                :if ($errorDescr ~ "^sftp-session-cut") do={
                    :error "break-loop"
                }
            }
        }

    } do={
        :if (!($errorDescr ~ "^break-loop")) do={
            :set state "Error When $state: $errorDescr"
            $globalNoteMe value=$state
            :if ([:len $firstError] = 0) do={ :set firstError $state }
            :set itsOk false
        }
    }
}

$globalNoteMe value="Housekeeping.." verbose=true
/file remove [/file find where name~$ramBackupPattern]
/file remove [/file find where name=$ramConfigLatest]

:local inf ""
:if ($itsOk) do={
    :set inf "✓ SUCCESS $scriptname on $sysname: automatic backup completed"
}

:if (!$itsOk) do={
    :set inf "✗ FAILED $scriptname on $sysname: $firstError"
}

$globalNoteMe value=$inf

:if (!$itsOk) do={
    $globalTgMessage value=$inf
    :error $inf
}
