:local sysname [/system identity get name]
:local scriptname "doBackup"
:global globalScriptBeforeRun
$globalScriptBeforeRun $scriptname

:global globalNoteMe
:global globalTgMessage
:global globalSecret
:global globalTimestamp
:global globalCallFetch
:global globalCallSSH
:global globalOnPrimaryPartition
:local itsOk true
:local state ""

:local rosVer [:tonum [:pick [/system resource get version] 0 1]]
:local sysver "NA"
:if ([:len [/system package find where name="system" and disabled=no]] > 0 and $rosVer = 6) do={
    :set sysver [/system package get system version]
}
:if ([:len [/system package find where name="routeros" and disabled=no]] > 0 and $rosVer = 7) do={
    :set sysver [/system package get routeros version]
}

:local useSysBackup true
:local useEncryption false
:local useRawExport true
:local useVerboseRawExport false

# directories have to exist!
:local useFtp true
:local ftpServer "$[$globalSecret get SSH_SERVER]"
:local ftpPort "$[$globalSecret get SSH_PORT]"
:local ftpUser "$[$globalSecret get SSH_USER]"
:local ftpPass "$[$globalSecret get SSH_PASSWORD]"
:local ftpRoot "REPO/backups/"
:local useFtpGit true
:local ftpRawGitName ("REPO/raw/rawconf_" . $sysname . "_latest.rsc")

:local sysnote [/system note get note]

:local stamp [$globalTimestamp]

:local useSmtp true
:local smtpAddress "defm.kopcap@gmail.com"
:local smtpSubject "$sysname Full Backup ($stamp)"
:local smtpBody "$sysname full Backup file see in attachment.\n $sysnote"

# preflight
:if (![$globalOnPrimaryPartition]) do={
    :set state "WARNING: the system booted up from fallback partition - skipping backup!"
    :log error $state
    $globalNoteMe value=$state
    :error $state
}

:onerror errorDescr in={
    :local smtpserv [:resolve "$ftpServer"]
} do={
    :set state "WARNING: FTP server '$ftpServer' unreachable - skipping backup!"
    :log error $state
    $globalNoteMe value=$state
    :error $state
}

:local fname "BACKUP-$sysname-$stamp"

:onerror errorDescr in={

    :if ($useSysBackup) do={
        :set state "Saving system backup $fname.backup"
        $globalNoteMe value=$state
        :if ($useEncryption) do={ /system backup save name=($fname . ".backup") }
        :if (!$useEncryption) do={ /system backup save dont-encrypt=yes name=($fname . ".backup") }
        :delay 2s
        $globalNoteMe value="System Backup Finished"
    }

    :if ($useRawExport and $useFtpGit) do={
        :set state "Exporting raw configuration $fname.safe.rsc"
        $globalNoteMe value=$state
        # show sensitive data
        :if ($useVerboseRawExport) do={ /export show-sensitive terse verbose file=($fname . ".safe.rsc") }
        :if (!$useVerboseRawExport) do={ /export show-sensitive terse file=($fname . ".safe.rsc") }
        :delay 2s
        $globalNoteMe value="Raw configuration script export Finished"
    }

    :delay 5s

    # make some backdoor
    :local wanIp [/ip cloud get public-address]
    :local remoteCommand "{ /ip/firewall/address-list remove [find list=alist-fw-knockknock address=$wanIp]; /ip/firewall/address-list/add address=$wanIp list=alist-fw-knockknock comment=doBackup timeout=5m dynamic=yes }"

    :set state "Making SFTP backdoor for $wanIp via SSH"
    $globalNoteMe value=$state

    :local errorDef [$globalCallSSH $remoteCommand]
    :if ([:len $errorDef] > 0) do={ :error $errorDef }

    :set state "Success RPC call: $state"
    $globalNoteMe value=$state

    # we have to wait while backdoor alist-fw-ssh-stage1 flushes our first-call Ip, otherwise we get banned
    :set state "Waiting backdoor for $wanIp for some seconds.."
    $globalNoteMe value=$state
    :delay 15s

} do={
    :set state "Error When $state: $errorDescr"
    $globalNoteMe value=$state
    :set itsOk false
}

:if ($itsOk) do={

    :foreach backupFile in=[/file find] do={

        :local buFile [/file get $backupFile name]

        :if ([:typeof [:find $buFile $fname]] != "nil") do={

            :local itsSrc ($buFile ~ ".safe.rsc")

            :if ($useFtp) do={

                :onerror errorDescr in={
                    :set state "Uploading $buFile to FTP ($ftpRoot$buFile)"
                    $globalNoteMe value=$state

                    :local dst "$ftpRoot$buFile"
                    :local fetchCmd "/tool fetch url=sftp://$ftpServer:$ftpPort/$dst src-path=$buFile user=$ftpUser password=$ftpPass upload=yes"
                    $globalCallFetch $fetchCmd

                    $globalNoteMe value="Done"
                } do={
                    :set state "Error When $state: $errorDescr"
                    $globalNoteMe value=$state
                    :set itsOk false
                }

                # special ftp upload for git purposes
                :if ($itsSrc and $useFtpGit) do={
                    :onerror errorDescr in={
                        :set state "Uploading $buFile to GIT-FTP (RAW, $ftpRawGitName)"
                        $globalNoteMe value=$state

                        :local dst "$ftpRawGitName"
                        :local fetchCmd "/tool fetch url=sftp://$ftpServer:$ftpPort/$dst src-path=$buFile user=$ftpUser password=$ftpPass upload=yes"
                        $globalCallFetch $fetchCmd

                        $globalNoteMe value="Done"
                    } do={
                        :set state "Error When $state: $errorDescr"
                        $globalNoteMe value=$state
                        :set itsOk false
                    }
                }
            }

            :if ($useSmtp and !$itsSrc) do={
                :onerror errorDescr in={
                    :set state "Uploading $buFile to SMTP"
                    $globalNoteMe value=$state

                    # email works in background, delay needed
                    /tool e-mail send to=$smtpAddress body=$smtpBody subject=$smtpSubject file=$buFile tls=starttls

                    # waiting for email to be delivered
                    :delay 15s

                    :if ([/tool e-mail get last-status] != "succeeded") do={
                        :error "e-mail last-status is not succeeded"
                    }

                    $globalNoteMe value="Done"
                } do={
                    :set state "Error When $state: $errorDescr"
                    $globalNoteMe value=$state
                    :set itsOk false
                }
            }

            :delay 2s
            /file remove $backupFile
        }
    }
}

:local inf ""
:if ($itsOk) do={
    :set inf "✓ SUCCESS $scriptname on $sysname: automatic backup completed"
}

:if (!$itsOk) do={
    :set inf "✗ FAILED $scriptname on $sysname: $state"
}

$globalNoteMe value=$inf

:if (!$itsOk) do={
    $globalTgMessage value=$inf
    :error $inf
}
