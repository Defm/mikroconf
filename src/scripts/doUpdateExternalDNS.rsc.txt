:local sysname [/system identity get name]
:local scriptname "doUpdateExternalDNS"
:global globalScriptBeforeRun
$globalScriptBeforeRun $scriptname

:global globalNoteMe
:global globalTgMessage
:global globalCallSSH
:global globalKnockKnock
:local itsOk true
:local payLoad false
:local state ""

# last seen external IP, survives between runs until doEnvironmentClearance
:global globalLastIpv4

# set true to refresh address-list/DNS/CHR even when IP did not change
:local useForcedRefresh false

:local ipv4 ""

:onerror errorDescr in={

    # parsing the current IPv4 result
    :set state "Requesting cloud-DNS public address"
    $globalNoteMe value=$state
    /ip cloud force-update
    :delay 7s
    :set ipv4 [/ip cloud get public-address]

    :if ([:typeof [:toip $ipv4]] != "ip") do={
        :error "No cloud-DNS IP received"
    }
    $globalNoteMe value="Done"

    :if (($globalLastIpv4 != $ipv4) or $useForcedRefresh) do={

        :set state "External IP changed: current - ($ipv4), last - ($globalLastIpv4)"
        $globalNoteMe value=$state

        /ip firewall address-list remove [find list~"alist-nat-external-ip"]
        /ip firewall address-list add list="alist-nat-external-ip" address=$ipv4

        /ip dns static remove [/ip dns static find name=ftpserver.org]
        /ip dns static add name=ftpserver.org address=$ipv4

        :set globalLastIpv4 $ipv4
        :set payLoad true

        # refresh CHR IPSEC preferred policy (was doSuperviseCHRviaSSH)
        :set state "Refreshing VPN server (CHR) IPSEC policies"
        $globalNoteMe value=$state

        # SSH RPCs look like bruteforce on CHR without knockknock
        :local errorDef [$globalKnockKnock $scriptname]
        :if ([:len $errorDef] > 0) do={ :error $errorDef }

        :local policyComment [/ip dns static get value-name=text [find where type=TXT and name="special-remote-CHR-ipsec-policy-comment"]]
        :local asyncCmd ":global globalPolicyComment $policyComment"

        :set state "Calling --- $asyncCmd"
        $globalNoteMe value=$state
        :set errorDef [$globalCallSSH $asyncCmd]
        :if ([:len $errorDef] > 0) do={ :error $errorDef }
        :set state "RPC: set remote preferred policy comment variable: ($policyComment) - Ok"
        $globalNoteMe value=$state

        :set asyncCmd ":global globalRemoteIp $ipv4/32"

        :set state "Calling --- $asyncCmd"
        $globalNoteMe value=$state
        :set errorDef [$globalCallSSH $asyncCmd]
        :if ([:len $errorDef] > 0) do={ :error $errorDef }
        :set state "RPC: set remote preferred policy IP variable: ($ipv4) - Ok"
        $globalNoteMe value=$state

        :set asyncCmd "/system script run doUpdatePoliciesRemotely"

        :set state "Calling --- $asyncCmd"
        $globalNoteMe value=$state
        :set errorDef [$globalCallSSH $asyncCmd]
        :if ([:len $errorDef] > 0) do={ :error $errorDef }
        :set state "RPC: call remote script: doUpdatePoliciesRemotely - Ok"
        $globalNoteMe value=$state
    }

} do={
    :set state "Error When $state: $errorDescr"
    $globalNoteMe value=$state
    :set itsOk false
}

:local inf ""
:if ($itsOk and $payLoad) do={
    :set inf "✓ SUCCESS $scriptname on $sysname: external IP address change detected, refreshed"
}

:if ($itsOk and !$payLoad) do={
    :set inf "✓ SUCCESS $scriptname on $sysname: no external IP address update needed"
}

:if (!$itsOk) do={
    :set inf "✗ FAILED $scriptname on $sysname: $state"
}

$globalNoteMe value=$inf

:if (!$itsOk) do={
    $globalTgMessage value=$inf
    :error $inf
}
