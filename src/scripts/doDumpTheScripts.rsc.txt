:local sysname [/system identity get name]
:local scriptname "doDumpTheScripts"
:global globalScriptBeforeRun
$globalScriptBeforeRun $scriptname

:global globalNoteMe
:global globalTgMessage
:global globalSecret
:global globalCallSftp
:global globalKnockKnock
:global globalOnPrimaryPartition
:global globalScriptId
:local itsOk true
:local state ""

:local ramDiskName "RAM"
:local ramSubDir "scripts/"

# paths — host /REPO/raw is the git clone watched by gitwatch (Docker)
:local ftpRoot "REPO"
:local ftpSubDir "raw/src/scripts/"
:local useFtp true
:local ftpServer "$[$globalSecret get SSH_SERVER]"

# sshfs disk dramatically SLOOOOOOWS any '/file find' operations
:local useSshfs false
:local sshDiskName "sshfs"
:local sshSubDir "raw/src/scripts/"

# predicate: true when a disk with given slot and type exists
:local checkDisk do={
    :local targetName $1
    :local targetType $2

    # both parameters are required
    :if (([:len $targetName] = 0) or ([:len $targetType] = 0)) do={ :return false }

    :foreach i in=[/disk find] do={
        :local dname [/disk get $i slot]
        :local dtype [/disk get $i type]
        :if (($dtype = $targetType) and ($dname = $targetName)) do={ :return true }
    }
    :return false
}

:local searchReplace do={
    :local input [:tostr $1]
    :local search [:tostr $2]
    :local replace [:tostr $3]

    :local start -1
    :local tmppos 0
    :local sx ""
    :local dx ""

    :while ([:typeof [:find $input $search $start]] = "num") do={
        :set tmppos [:find $input $search $start]
        :set sx [:pick $input 0 $tmppos]
        :set dx [:pick $input ($tmppos + [:len $search]) [:len $input]]
        :set start ([:len "$sx$replace"] - 1)
        :set input "$sx$replace$dx"
    }
    :return $input
}

# preflight
:if (![$globalOnPrimaryPartition]) do={
    :set state "WARNING: the system booted up from fallback partition - skipping dump!"
    :log error $state
    $globalNoteMe value=$state
    :error $state
}

:if (![$checkDisk $ramDiskName "tmpfs"]) do={
    :set state "WARNING: tmpfs '$ramDiskName' not found - skipping dump!"
    :log error $state
    $globalNoteMe value=$state
    :error $state
}

:if ($useSshfs and ![$checkDisk $sshDiskName "sshfs"]) do={
    :set state "WARNING: sshfs '$sshDiskName' not found - skipping dump!"
    :log error $state
    $globalNoteMe value=$state
    :error $state
}

:if ($useFtp) do={
    :onerror errorDescr in={
        :local ftpIp [:resolve $ftpServer]
    } do={
        :set state "WARNING: FTP server '$ftpServer' unreachable - skipping dump!"
        :log error $state
        $globalNoteMe value=$state
        :error $state
    }
}

:local dumpPath "$ramDiskName/$ramSubDir"

$globalNoteMe value="Clearing RAMdisk.."
/file remove [/file find where name~"$dumpPath"]

$globalNoteMe value="Scripts source export.."

:foreach scriptId in=[/system script find] do={

    :onerror errorDescr in={

        :local scriptSource [/system script get $scriptId source]
        :local scriptName [/system script get $scriptId name]
        :local scriptSize [:len $scriptSource]

        :local scriptPath "$dumpPath$scriptName.rsc.txt"

        :set globalScriptId $scriptId

        :if ($scriptSize >= 4096) do={
            :set state "Please keep care about '$scriptName' consistency - its size over 4096 bytes"
            $globalNoteMe value=$state
        }

        # remove stale file, then create empty one
        /file remove [find where name=$scriptPath type!=directory]
        /file add name=$scriptPath contents=""

        :set state "Exporting '$scriptName' to '$scriptPath'"
        $globalNoteMe value=$state

        /execute script=":global globalScriptId; :put [/system script get $globalScriptId source];" file=$scriptPath

        $globalNoteMe value="Done"

    } do={
        :set state "Error When $state: $errorDescr"
        $globalNoteMe value=$state
        :set itsOk false
    }
}

# lots of tiny SFTP uploads look like bruteforce on CHR without knockknock
:local errorDef [$globalKnockKnock $scriptname]
:if ([:len $errorDef] > 0) do={
    :set state $errorDef
    $globalNoteMe value=$state
    :set itsOk false
}

$globalNoteMe value="Scripts source pushing.."

:if ($itsOk) do={

    :local stopBatch false

    :foreach backupFile in=[/file find where name~"$dumpPath"] do={

        :if (!$stopBatch) do={

            :local scriptPath [/file get $backupFile name]

            :if ([:typeof [:find $scriptPath ".rsc.txt"]] != "nil") do={

                # FTP upload
                :if ($useFtp) do={

                    :local dst [$searchReplace $scriptPath $ramDiskName $ftpRoot]
                    :set dst [$searchReplace $dst $ramSubDir $ftpSubDir]

                    :onerror errorDescr in={

                        :set state "Uploading $scriptPath to SFTP - $dst"
                        $globalNoteMe value=$state

                        :local errorDef [$globalCallSftp $scriptPath $dst]
                        :if ([:len $errorDef] > 0) do={ :error $errorDef }

                        $globalNoteMe value="Done"

                    } do={
                        :set state "Error When $state: $errorDescr"
                        $globalNoteMe value=$state
                        :set itsOk false
                        :if ($errorDescr ~ "^sftp-session-cut") do={
                            :set stopBatch true
                        }
                    }
                }

                # SSHFS copy
                :if ($useSshfs) do={

                    :local dst [$searchReplace $scriptPath $ramDiskName $sshDiskName]
                    :set dst [$searchReplace $dst $ramSubDir $sshSubDir]

                    :onerror errorDescr in={

                        :set state "Clearing $dst on SSH-disk"
                        $globalNoteMe value=$state
                        /file remove [/file find where name=$dst type!=directory]

                        :set state "Copying $scriptPath to SSH-disk - $dst"
                        $globalNoteMe value=$state
                        /file copy $scriptPath name=$dst

                        $globalNoteMe value="Done"

                    } do={
                        :set state "Error When $state: $errorDescr"
                        $globalNoteMe value=$state
                        :set itsOk false
                    }
                }
            }
        }
    }
}

:delay 5s

$globalNoteMe value="Housekeeping.."
/file remove [/file find where name~"$dumpPath"]

:local inf ""
:if ($itsOk) do={
    :set inf "✓ SUCCESS $scriptname on $sysname: scripts dumped"
}

:if (!$itsOk) do={
    :set inf "✗ FAILED $scriptname on $sysname: $state"
}

$globalNoteMe value=$inf

:if (!$itsOk) do={
    $globalTgMessage value=$inf
    :error $inf
}
