:local sysname [/system identity get name]
:local scriptname "doHeatFlag"
:global globalScriptBeforeRun
$globalScriptBeforeRun $scriptname

:global globalNoteMe
:global globalTgMessage
:local itsOk true
:local state ""

# cpu temperature threshold, C
:local maxTemp 68

:onerror errorDescr in={

    :set state "Checking cpu temperature against $maxTemp C"
    $globalNoteMe value=$state
    :local currentTemp [/system/health get [find name=cpu-temperature] value]

    :if ($currentTemp > $maxTemp) do={

        :set state "system overheat at $currentTemp C"
        $globalNoteMe value=$state
        :set itsOk false

        /beep length=.1
        :delay 250ms
        /beep length=.1
        :delay 800ms
        /beep length=.1
        :delay 250ms
        /beep length=.1
        :delay 800ms

    } else={
        :set state "cpu temperature ok ($currentTemp C)"
        $globalNoteMe value=$state
    }

} do={
    :set state "Error When $state: $errorDescr"
    $globalNoteMe value=$state
    :set itsOk false
}

:local inf ""
:if ($itsOk) do={
    :set inf "✓ SUCCESS $scriptname on $sysname: $state"
}

:if (!$itsOk) do={
    :set inf "✗ FAILED $scriptname on $sysname: $state"
}

$globalNoteMe value=$inf

:if (!$itsOk) do={
    $globalTgMessage value=$inf
    :error $inf
}
