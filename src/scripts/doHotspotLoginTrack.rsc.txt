# fill it inside hotspot on-login
:global globalHotspotUser

# may be triggered twice per moment - snapshot globals first
:local hotspotUser $globalHotspotUser

# init globals as far as we are inside *sys user account
:global globalSafeScriptCall
:if (!any $globalSafeScriptCall) do={ /system script run doEnvironmentSetup }

:local sysname [/system identity get name]
:local scriptname "doHotspotLoginTrack"
:global globalScriptBeforeRun
$globalScriptBeforeRun $scriptname "HOTSPOT"

:global globalNoteMe
:global globalTgMessage
:local itsOk true
:local state ""

:if (!any $hotspotUser) do={
    :set state "No globalHotspotUser provided.."
    $globalNoteMe value=$state
    :error $state
}

:onerror errorDescr in={

    :set state "Tracking hotspot login for $hotspotUser"
    $globalNoteMe value=$state

    :local today [/system clock get date]
    :local time1 [/system clock get time]
    :local ipuser [/ip hotspot active get [find user=$hotspotUser] address]
    :local usermac [/ip hotspot active get [find user=$hotspotUser] mac-address]

    :local hour [:pick $time1 0 2]
    :local min [:pick $time1 3 5]
    :local sec [:pick $time1 6 8]
    :set time1 ($hour . $min . $sec)

    :local mac1 [:pick $usermac 0 2]
    :local mac2 [:pick $usermac 3 5]
    :local mac3 [:pick $usermac 6 8]
    :local mac4 [:pick $usermac 9 11]
    :local mac5 [:pick $usermac 12 14]
    :local mac6 [:pick $usermac 15 17]
    :set usermac ($mac1 . $mac2 . $mac3 . $mac4 . $mac5 . $mac6)

    /ip firewall address-list add list=$today address=("log-in." . $time1 . "." . $hotspotUser . "." . $usermac . "." . $ipuser)

    :set state "Logged hotspot login $hotspotUser at $time1"
    $globalNoteMe value=$state
    $globalNoteMe value="Done"

} do={
    :set state "Error When $state: $errorDescr"
    $globalNoteMe value=$state
    :set itsOk false
}

:local inf ""
:if ($itsOk) do={
    :set inf "✓ SUCCESS $scriptname on $sysname: $state"
}

:if (!$itsOk) do={
    :set inf "✗ FAILED $scriptname on $sysname: $state"
}

$globalNoteMe value=$inf

:if (!$itsOk) do={
    $globalTgMessage value=$inf
    :error $inf
}
